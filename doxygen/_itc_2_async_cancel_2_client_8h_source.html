28/** This concrete class is a asynchronous ITC client that cycles through sending
29 different flash duty cycle requests to the Server that controls the LED
30 flashing
31 */
32class Client : public IFlashResponse, public Kit::Itc::OpenCloseSync
33{
34public:
35 /// Constructor
36 Client( Kit::EventQueue::IQueue& clientsEventQueue,
37 IFlashRequest::SAP& serverSAP,
38 FlashPattern_T flashPatterns[],
39 unsigned numFlashPatterns,
40 Kit::System::Semaphore& shutdownSignal )
41 : Kit::Itc::OpenCloseSync( clientsEventQueue )
42 , m_shutdownSignal( shutdownSignal )
43 , m_serverSAP( serverSAP )
44 , m_flashResponseMsg( *this, clientsEventQueue, serverSAP, m_flashPayload )
45 , m_flashCancelResponseMsg( *this, clientsEventQueue, serverSAP, m_flashCancelPayload )
46 , m_flashPatterns( flashPatterns )
47 , m_numFlashPatterns( numFlashPatterns )
48 , m_maxNumberOfRequests( 0 )
49 , m_opened( false )
50 {
51 KIT_SYSTEM_ASSERT( m_flashPatterns != nullptr );
52 }
53
54public:
55 /// See Kit::Itc::OpenCloseSync
56 void request( OpenMsg& msg ) noexcept override
57 {
58 // Only 'open' when not already opened
59 if ( !m_opened )
60 {
61 // Housekeeping
62 m_opened = true;
63 m_flashPatternIndex = 0;
64 m_closedMsg = nullptr;
65
66 // Get the maximum number of requests from the message payload
67 m_maxNumberOfRequests = static_cast<int>( reinterpret_cast<size_t>( msg.getPayload().args ) );
68
69 // Send the first server request
70 sendRequest();
71 }
72
73 // Return the message to the sender since we are done processing the request
74 msg.returnToSender();
75 }
76
77 /// See Kit::Itc::OpenCloseSync
78 void request( CloseMsg& msg ) noexcept override
79 {
80 // Only 'close' when already opened
81 if ( m_opened )
82 {
83 // Housekeeping
84 m_opened = false;
85
86 // Unconditionally cancel any pending flash request before closing
87 KIT_SYSTEM_TRACE_MSG( "main", "Cancelling any pending flash request before closing." );
88 m_closedMsg = &msg;
89 m_flashCancelPayload.flashMsgToCancel = &m_flashResponseMsg.getRequestMsg();
90 m_serverSAP.post( m_flashCancelResponseMsg.getRequestMsg() );
91 return; // The Close message will be returned once the cancel response is received
92 }
93
94 // Return the message to the sender since we are done processing the request
95 msg.returnToSender();
96 }
97
98public:
99 /// Flash Response
100 void response( FlashMsg& msg ) noexcept override
101 {
102 // Ignore the response if the client is in the process of closing
103 if ( m_closedMsg )
104 {
105 KIT_SYSTEM_TRACE_MSG( "main", "Ignoring flash response since client is closing." );
106 return;
107 }
108
109 // Process the Response
110 auto& payload = msg.getPayload();
111 if ( payload.success )
112 {
113 // Limit the number of requests
114 if ( --m_maxNumberOfRequests > 0 )
115 {
116 sendRequest();
117 }
118
119 // Shutdown the application when the maximum number of requests has been reached
120 else
121 {
122 KIT_SYSTEM_TRACE_MSG( "main", "Maximum number of requests reached." );
123 m_shutdownSignal.signal();
124 }
125 }
126
127 // End the application on error
128 else
129 {
130 KIT_SYSTEM_TRACE_MSG( "main", "Flash Request FAILED. (idx=%" PRIu32 ")", m_flashPatternIndex );
131 m_shutdownSignal.signal();
132 }
133 }
134
135 /// Cancel Response
136 void response( FlashCancelMsg& msg ) noexcept override
137 {
138 // Process the Cancel Response
139 auto& payload = msg.getPayload();
140 if ( payload.canceled )
141 {
142 KIT_SYSTEM_TRACE_MSG( "main", "Flash Cancel Request: Request cancelled." );
143 }
144 else
145 {
146 KIT_SYSTEM_TRACE_MSG( "main", "Flash Cancel Request: No pending request to cancel." );
147 }
148
149 // End the close() process by returning the Close msg
150 if ( m_closedMsg )
151 {
152 KIT_SYSTEM_TRACE_MSG( "main", "Flash Client Closed." );
153 m_closedMsg->returnToSender();
154 }
155 }
156
157protected:
158 /// Helper method: Send the rate request to the server
159 void sendRequest() noexcept
160 {
161 // Buid the Request payload
162 auto& payload = m_flashResponseMsg.getPayload();
163 payload.flashPattern = m_flashPatterns[m_flashPatternIndex];
164 KIT_SYSTEM_TRACE_MSG( "main",
165 "Sending Rate Request (on=%" PRIu32 " ms, off=%" PRIu32 " ms, n=%" PRIu32 ")",
166 payload.flashPattern.onTimeMs,
167 payload.flashPattern.offTimeMs,
168 payload.flashPattern.repeatCount );
169
170 // Increment the flash rate index for the next request
171 incrementFlashRateIndex();
172
173 // Send the request to the server
174 m_serverSAP.post( m_flashResponseMsg.getRequestMsg() );
175 }
176
177 /// Helper
178 void incrementFlashRateIndex() noexcept
179 {
180 m_flashPatternIndex = ( m_flashPatternIndex + 1 ) % m_numFlashPatterns;
181 }
182
183protected:
184 /// Used to signal the main thread that the client has finished its work.
185 Kit::System::Semaphore& m_shutdownSignal;
186
187 /// Server Widget's SAP (aka its fully qualified 'handle')
188 IFlashRequest::SAP& m_serverSAP;
189
190 /// Flash Msg: Payload
191 IFlashRequest::FlashPayload_T m_flashPayload;

193 /// Flash Msg: Response message (which contains the Request message)

195
196 /// CANCEL Flash Msg: Payload
197 IFlashRequest::FlashCancelPayload_T m_flashCancelPayload;
198
199 /// CANCEL Flash Msg: Response message (which contains the Request message)
200 IFlashResponse::FlashCancelMsg m_flashCancelResponseMsg;
201
202 /// Cached closed msg
203 CloseMsg* m_closedMsg;
204
205 /// Flash rate table
206 FlashPattern_T* m_flashPatterns;
207
208 /// Index of the current flash rate
209 uint32_t m_flashPatternIndex;
210
211 /// Flash rate table count
212 unsigned m_numFlashPatterns;
213
214 /// Maximum number of requests to send

217 /// Track my open/closed
218 bool m_opened;
