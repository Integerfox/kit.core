26/** This class implements the Non-volatile storage driver using a collection
27 of NV drivers to provide a 'flat address' space that spans multiple
28 discrete storage chips/media.
29
30 The individual drivers/nv-storage-instance can be different sizes and
31 different physical media.
32
33 The interface itself is NOT thread safe. It is the responsibility of
34 the users/clients of the driver to handle any threading issues.
35 */
36class Gang : public IApi
37{
38public:
39 /** Constructor. The 'drivers' argument is an array of pointers to concrete
40 NV drivers and 'numDrivers' is the number of elements in the array.
41 The array must contain at least one driver and none of the array
42 elements can be a nullptr (the constructor asserts on violations).
43 The array is NOT copied, i.e. it must remain valid for the lifetime
44 of the Gang instance. The constructor does not access the drivers
45 themselves.
46 */
47 Gang( IApi* drivers[], unsigned numDrivers ) noexcept;
48
49public:
50 /** See Kit::Driver::IStart. The 'startArgs' argument is passed to the
51 start() method of each of the ganged drivers.
52 */
53 bool start( void* startArgs = nullptr ) noexcept override;
54
55 /// See Kit::Driver::IStop
56 void stop() noexcept override;
57
58public:
59 /// See Kit::Driver::NV::IApi
60 bool write( size_t dstOffset, const void* srcData, size_t numBytesToWrite ) noexcept override;
61
62 /// See Kit::Driver::NV::IApi
63 bool read( size_t srcOffset, void* dstData, size_t sizeDstData, size_t numBytesToRead ) noexcept override;
64
65 /** See Kit::Driver::NV::IApi. The number of pages is based on the smallest
66 page size from the various ganged drivers that fits within the total
67 available space. When the ganged drivers have pages size that are not
68 multiples of each other, the returned value can be just approximate
69 number of pages.
70 */
71 size_t getNumPages() const noexcept override;
72
73 /** See Kit::Driver::NV::IApi. The reported page size is the smallest page

75 demand, i.e. it is not cached, and is only valid while the instance
76 is started (returns zero when the instance is stopped).
77 */
78 size_t getPageSize() const noexcept override;

83 size_t getTotalSize() const noexcept override;
