25/** This Private Namespace abstract class defines a "Context" for a TShell
26 command. The Context provide common infrastructure, information, buffers,
27 etc. that facilitates interaction between the ICommand Processor and individual
28 commands. The application SHOULD NEVER directly access this interface.
29*/
30class IContext
31{
32public:
33 /// This method returns the list of implemented commands
34 virtual Kit::Container::OrderedList<ICommand>& getCommands() noexcept = 0;
35
36 /// Lookup a command by its verb. Returns a nullptr if the command is not found
37 virtual ICommand* findCommand( const char* verb, unsigned verbLength ) noexcept = 0;
38
39public:
40 /** This method encodes and outputs the specified message/text. The method
41 returns false if there was Output Stream error.
42
43 NOTE: An ENTIRE frame must be written with a single call to this method
44 to avoid interleaving of output frames with console trace/logging.
45 */
46 inline bool writeFrame( const char* text ) noexcept
47 {
48 return writeFrame( text, strlen( text ) );
49 }
50
51 /** Same as writeFrame(), but only outputs (at most) 'N' bytes as the content
52 of the frame
53 */
54 virtual bool writeFrame( const char* text, Kit::Type::SSize_T maxBytes ) noexcept = 0;
55
56public:
57 /** This method returns a working buffer for a command to format its
58 output prior to 'writing the frame'. The buffer is only 'valid' for
59 duration of the command's execute() method. The string buffer is
60 guaranteed to be at least size of the largest supported output frame.
61 */
62 virtual Kit::Text::IString& getOutputBuffer() noexcept = 0;
63
64 /** This method returns a 'working' buffer for use by a command when
65 parsing input arguments. The buffer is only 'valid' for
66 duration of the command's execute() method. The string buffer is
67 guaranteed to be at least size of the largest supported input frame.
68 */
69 virtual Kit::Text::IString& getInputBuffer() noexcept = 0;
70
71 /** This method returns a 'working' buffer for use by a command. The buffer
72 is only 'valid' for duration of the command's execute() method. The
73 buffer is guaranteed to be at least size the same size as the 'Input
74 buffer'
75 */
76 virtual Kit::Text::IString& getWorkBuffer() noexcept = 0;
77
78public:
79 /** This method allows a command access to the input stream, i.e. consume
80 additional input data that is 'out-of-band' of the nominal newline
81 delineated command syntax.
82
83 Attempts to read the specified number of bytes from the stream into the
84 supplied buffer. The actual number of bytes read is returned via
85 'bytesRead'. Returns true if successful, or false if End-of-Stream
86 was encountered.
87 */
88 virtual bool oobRead( void* dstBuffer,
89 Kit::Type::SSize_T numBytes,
90 Kit::Type::SSize_T& bytesRead ) noexcept = 0;
91
92public:
93 /// Returns reference to the Processor's Security instance
94 virtual ISecurity& getSecurity() noexcept = 0;
95
96public:
97 /** Returns a pointer to the Processor's output stream. If the TShell is
98 has not been started, then nullptr is returned.
99 */
100 virtual Kit::Io::IOutput* getOutputStream() noexcept = 0;
101
102public:
103 /** Request to terminated/exit the TShell. When/if the TShell is exited
104 is application specific behavior.
105 */
106 virtual void requestTShellExit() noexcept = 0;
107
108public:
109 /// Virtual destructor
110 virtual ~IContext() = default;
111};
