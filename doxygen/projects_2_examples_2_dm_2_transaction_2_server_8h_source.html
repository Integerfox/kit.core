28/** This concrete class is a 'server' that responds to requests from a 'client'
29 to generate. See the README.md file for more details
30 */
31class Server : public Kit::Itc::OpenCloseSync
32{
33public:
34 /// Constructor
35 Server( Kit::EventQueue::IQueue& myMbox,
36 MpFoo& mpFoo )
37 : Kit::Itc::OpenCloseSync( myMbox )
38 , m_mpFoo( mpFoo )
39 , m_obFoo( myMbox )
40 , m_opened( false )
41 {
42 m_obFoo.setCallback<Server, &Server::triggerChanged>( this );
43 }
44
45public:
46 /// See Kit::Itc::OpenCloseSync
47 void request( OpenMsg& msg ) noexcept override
48 {
49 // Only 'open' when not already opened
50 if ( !m_opened )
51 {
52 // Housekeeping
53 m_opened = true;
54
55 // Subscribe for the 'trigger' MP change notification.
56 m_mpFoo.attach( m_obFoo );
57 }
58
59 // Return the message to the sender since we are done processing the request
60 msg.returnToSender();
61 }
62
63 /// See Kit::Itc::OpenCloseSync
64 void request( CloseMsg& msg ) noexcept override
65 {
66 // Only 'close' when already opened
67 if ( m_opened )
68 {
69 // Housekeeping
70 m_opened = false;
71
72 // Make sure the all of the potential callbacks get cancelled
73 m_mpFoo.detach( m_obFoo );
74 }
75
76 // Return the message to the sender since we are done processing the request
77 msg.returnToSender();
78 }
79
80
81protected:
82 /// Callback when the 'trigger' MP changes
83 void triggerChanged( MpFoo& mpThatChanged, Kit::Dm::IObserver& observer ) noexcept
84 {
85 Foo_T data;
86 if ( mpThatChanged.readAndSync( data, observer ) )
87 {
88 // 'Alive' feedback when running on hardware
89 Bsp_toggle_debug1();
90
91 // Generate a Random number between the upper and lower limits and write to the MP
92 // TODO: Need a Random Number interface
93 data.randomNumber = ( ( data.upperLimit - data.lowerLimit ) / 2 ) + data.lowerLimit;
94 data.isValid = true; // For this example - the result is always valid (because I am lazy)
95 KIT_SYSTEM_TRACE_MSG( "main",
96 "SERVER: request. upper=%" PRId32 ", lower=%" PRId32 ", result=%" PRId32 ", isValid=%s",
97 data.upperLimit,
98 data.lowerLimit,
99 data.randomNumber,
100 data.isValid ? "true" : "false" );
101
102 // Update the MP with response AND make sure that a change notification
103 // gets generated (i.e. the 'true' argument). Why is the 'force'
104 // required? It's because change notifications are only generated when
105 // the MP value is changed. It is theoretically possible that the
106 // server's random-number/valid-invalid state values do NOT change
107 // the MP actual value. Hence, the 'force' argument is used to
108 // make sure that the Client gets notified of the response.
109 mpThatChanged.detach( m_obFoo ); // Detach so we don't get callback for WRITE
110 uint16_t seqNum = mpThatChanged.write( data, true );
111 mpThatChanged.attach( m_obFoo, seqNum ); // Re-attach for the next request
112 }
113 }
114
115protected:
116 /// MP for the Client/Server transaction
117 MpFoo& m_mpFoo;
118
119 /// Observer callback for the MP change notification
120 Kit::Dm::ObserverCallback<MpFoo> m_obFoo;
121
122 /// Track my open/closed
123 bool m_opened;
124};
