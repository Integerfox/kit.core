46/** This class is an Event Queue (Kit::EventQueue::Server) that is NOT run by
47 a thread. Instead, the unit test runs it - on the test's own thread - by
48 calling runUntilIdle(). The test therefore controls exactly when the code
49 that executes in the Event Queue's context (ITC requests, Model Point
50 change notifications, event flag handlers, software timer callbacks) runs -
51 and, for software timers, how much time has passed (see advanceTime()).
52 This makes tests deterministic: once runUntilIdle() returns true, the
53 queue's servers have processed everything they are going to process, i.e.
54 'nothing happened' checks need no sleeps, and 'something happened' checks
55 need no polling/timeouts.
56
57 How to use:
58 1. Construct a ManualEventQueue instead of a Kit::EventQueue::Server,
59 and pass it to the object(s) under test as their Event Queue. Do
60 NOT create a thread for it.
61 2. Open/close servers with requestOpen()/requestClose() - NOT with the
62 server's synchronous open()/close() methods (see the 'Limitations'
63 below). Other ITC requests are sent with postAndRun().
64 3. After every stimulus (e.g. writing a Model Point, adding an entry
65 to a RingBufferMP), call runUntilIdle() and then check the results.
66 Wrap the call in REQUIRE() - a false return value means the
67 server never became idle (i.e. it is busy-looping).
68
69 Example (a server that implements Kit::Itc::OpenCloseSync, and processes
70 Model Point change notifications):
71 \code
72
73 #include "Kit/_support/testing/ManualEventQueue.h"
74
75 TEST_CASE( "MyServer" )
76 {
77 ManualEventQueue eventQueue; // No thread!
78 MyServer uut( eventQueue, mp_input_ );
79
80 REQUIRE( eventQueue.requestOpen( uut ) ); // Instead of uut.open()
81
82 mp_input_.write( 42 ); // Stimulus...
83 REQUIRE( eventQueue.runUntilIdle() ); // ...let the server react...
84 REQUIRE( uut.m_count == 1 ); // ...and check the exact result
85
86 mp_input_.write( 42 ); // No change -->no notification
87 REQUIRE( eventQueue.runUntilIdle() );
88 REQUIRE( uut.m_count == 1 ); // Deterministic 'nothing happened' check
89
90 REQUIRE( eventQueue.requestClose( uut ) ); // Instead of uut.close()
91 }
92
93 \endcode
94
95 Example (any other ITC request - the result is read from the payload after
96 the queue has processed the message):
97 \code
98
99 MyServer::MyPayload_T payload( ... );
100 MyServer::MyMsg msg( uut, payload, eventQueue.nullReturnHandler() );
101 REQUIRE( eventQueue.postAndRun( msg ) );
102 REQUIRE( payload.m_success );
103
104 \endcode
105
106 Example (software timers - time only advances when the test says so):
107 \code
108
109 ManualEventQueue eventQueue;
110 MyWidget uut( eventQueue ); // Starts a 100 ms timer (Kit::System::Timer) on the queue
111
112 REQUIRE( eventQueue.advanceTime( 99 ) ); // 99 ms later...
113 REQUIRE( uut.m_timeoutCount == 0 ); // ...not expired
114 REQUIRE( eventQueue.advanceTime( 1 ) ); // 100 ms...
115 REQUIRE( uut.m_timeoutCount == 1 ); // ...expired, exactly once
116 REQUIRE( eventQueue.getElapsedTime() == 100 );
117
118 \endcode
119
120 Example (event flags - the handlers are notified by runUntilIdle()):
121 \code
122
123 Kit::Container::SList<Kit::System::IEventFlag> eventList;
124 MyEventFlagHandler handler( eventList, 1u << 3 ); // A Kit::System::EventFlagBase child
125 ManualEventQueue eventQueue( &eventList );
126
127 eventQueue.signalEvent( 3 ); // Stimulus (e.g. from the code under test)...
128 REQUIRE( handler.m_count == 0 ); // ...nothing happens until the queue runs
129 REQUIRE( eventQueue.runUntilIdle() );
130 REQUIRE( handler.m_count == 1 ); // Notified exactly once
131
132 \endcode
133
134 Limitations:
135 - Synchronous ITC calls (e.g. Kit::Itc::OpenCloseSync::open()/close(),
136 or any other xxxSync API) block until a server thread processes the
137 message - there is none, i.e. the call never returns. Use
138 requestOpen(), requestClose(), or postAndRun() instead.
139 - Messages are sent with a 'null' return handler, i.e. the server's
140 returnToSender() does nothing. Read the results from the payload
141 after the call.
142 - Software timers (Kit::System::Timer and friends that use the queue
143 as their Kit::System::ITimingSource) run on the queue's own virtual
144 clock, which starts at zero and ONLY advances with advanceTime().
145 Code that reads the system clock directly (e.g.
146 Kit::System::ElapsedTime) still sees real time.
147 - The queue must only be run by ONE thread (the test thread). Code
148 under test that creates its own threads is not made deterministic.
149
150 NOTE: The class is intended for unit tests only.
151 */
152class ManualEventQueue : public Kit::EventQueue::Server
153{
154public:
155 /** Constructor. The optional 'eventFlagsList' is the list of event flag
156 handlers that are notified (by runUntilIdle()) when an event flag is
157 signaled. See Kit::EventQueue::Server
158 */
159 explicit ManualEventQueue( Kit::Container::SList<Kit::System::IEventFlag>* eventFlagsList = nullptr ) noexcept
160 : Kit::EventQueue::Server( OPTION_KIT_SYSTEM_EVENT_LOOP_TIMEOUT_PERIOD, eventFlagsList )
161 , m_elapsedMs( 0 )
162 {
163 }
164
165public:
166 /** Processes the signaled event flags, pending ITC messages, and change
167 notifications until there is no pending work, or 'maxIterations'
168 iterations have been performed. Each iteration notifies the event
169 flag handlers first (the same order as a running Event Queue).
170 Returns true when the queue is idle, i.e. false means the server(s)
171 never stopped generating work (e.g. busy-looping). The number of
172 iterations performed is optionally returned via 'retIterations'.
173 */
174 bool runUntilIdle( unsigned maxIterations = OPTION_KIT_TESTING_MANUAL_EVENT_QUEUE_MAX_ITERATIONS,
175 unsigned* retIterations = nullptr ) noexcept
176 {
177 unsigned iterations = 0;
178 while ( iterations < maxIterations && hasPendingWork() )
179 {
180 processEventFlags();
181 serverProcessEvents();
182 iterations++;
183 }
184 if ( retIterations )
185 {
186 *retIterations = iterations;
187 }
188 return !hasPendingWork();
189 }
190
191 /// Returns true if there are signaled event flags that have not been processed
192 bool isPendingEventFlags() noexcept
193 {
194 Kit::System::GlobalLock::begin();
195 bool pending = m_events != 0;
196 Kit::System::GlobalLock::end();
197 return pending;
198 }
199
200 /// Returns true if there are signaled event flags, ITC messages and/or change notifications waiting to be processed
201 bool hasPendingWork() noexcept
202 {
203 return KIT_TESTING_MANUAL_EVENT_QUEUE_HAS_PENDING_WORK_();
204 }
205
206public:
207 /** Posts 'msg' (asynchronously) and runs the queue until idle. Returns
208 the result of runUntilIdle(). The message must be constructed with
209 nullReturnHandler() (or another non-blocking return handler).
210 */
211 bool postAndRun( Kit::Itc::IMessage& msg ) noexcept
212 {
213 post( msg );
214 return runUntilIdle();
215 }
216
217 /** Opens 'server' - the asynchronous replacement for the server's
218 synchronous open() method. Returns the server's open result, i.e.
219 false when the server fails to open, or the queue does not become idle.
220 */
221 bool requestOpen( Kit::Itc::IOpenRequest& server, void* args = nullptr ) noexcept
222 {
223 Kit::Itc::IOpenRequest::OpenPayload_T payload( args );
224 Kit::Itc::IOpenRequest::OpenMsg msg( server, payload, m_nullReturnHandler );
225 return postAndRun( msg ) && payload.success;
226 }
227
228 /** Closes 'server' - the asynchronous replacement for the server's
229 synchronous close() method. Returns the server's close result, i.e.
230 false when the server fails to close, or the queue does not become
231 idle.
232 */
233 bool requestClose( Kit::Itc::ICloseRequest& server, void* args = nullptr ) noexcept
234 {
235 Kit::Itc::ICloseRequest::ClosePayload_T payload( args );
236 Kit::Itc::ICloseRequest::CloseMsg msg( server, payload, m_nullReturnHandler );
237 return postAndRun( msg ) && payload.success;
238 }
239
240 /// Returns a return handler for messages that are sent with postAndRun()
241 Kit::Itc::IReturnHandler& nullReturnHandler() noexcept
242 {
243 return m_nullReturnHandler;
244 }
245
246public:
247 /** Advances the queue's virtual clock by 'milliseconds', one millisecond
248 at a time. For each millisecond, the expired software timers' callbacks
249 are called, and then the queue is run until idle (i.e. the work that
250 the callbacks generate is processed before the clock advances again -
251 the same order as a running Event Queue). A periodic timer, i.e. one
252 that restarts itself from its expired callback, therefore expires
253 once per period. Pending work is processed before the clock is
254 advanced. Returns false (immediately) if the queue does not become
255 idle.
256 */
257 bool advanceTime( uint32_t milliseconds ) noexcept
258 {
259 if ( !runUntilIdle() )
260 {
261 return false;
262 }
263 for ( uint32_t i = 0; i < milliseconds; i++ )
264 {
265 // Advance the clock first, i.e. the expired callbacks see the time at which they expire
266 m_elapsedMs++;
267 tick( 1 );
268 tickComplete();
269 if ( !runUntilIdle() )
270 {
271 return false;
272 }
273 }
274 return true;
275 }
276
277 /// Returns the queue's virtual clock, i.e. the total time advanced with advanceTime(), in milliseconds
278 uint64_t getElapsedTime() const noexcept
279 {
280 return m_elapsedMs;
281 }
282
283 /** See Kit::System::ITimingSource. One count is one millisecond of the
284 virtual clock, i.e. the timer duration does NOT include the (real) time
285 since the timers were last processed (the base class behavior).
286 */
287 uint32_t msecToCounts( uint32_t durationInMsecs ) const noexcept override
288 {
289 return durationInMsecs;
290 }
291
292protected:
293 /// Return handler for the messages sent by the test (they are never returned)
294 Kit::Itc::NullReturnHandler m_nullReturnHandler;
295
296 /// Virtual clock (milliseconds)
297 uint64_t m_elapsedMs;
298};
