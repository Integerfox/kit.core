25/** This class defines the interface for a digital output pin. The interface
26 operates on the LOGICAL state of the output, abstracting away the physical
27 polarity (active-high vs active-low). When the output is 'asserted', the
28 pin is in its logically active state regardless of the underlying electrical
29 polarity.
30
31 The interface itself is NOT thread safe. It is the responsibility of
32 the users/clients of the driver to handle any threading issues.
33 */
34class IOutput : public virtual Kit::Driver::IStart,
35 public virtual Kit::Driver::IStop
36{
37public:
38 /// Asserts (logically activates) the output
39 virtual void assertPin() noexcept = 0;
40
41 /// Deasserts (logically deactivates) the output
42 virtual void deassertPin() noexcept = 0;
43
44 /// Toggles (inverts) the current logical output state
45 virtual void toggle() noexcept = 0;
46
47 /// Sets the logical output state (true=asserted, false=deasserted)
48 virtual void set( bool asserted ) noexcept = 0;
49
50 /// Returns the current logical output state (true=asserted, false=deasserted)
51 virtual bool isAsserted() const noexcept = 0;
52
53
54public:
