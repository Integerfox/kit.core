29/** This abstract class defines the interface for a Job. An Job is essentially a
30 'mini-application' that performs a series of steps and typically reports its
31 output using the KIT_SYSTEM_TRACE_MSG() method.
32
33 Multiple Job instances can be executing at the same time. Specifically, one
34 or more Job instances execute in their IManager's threads. Job self register
35 with their IManager. Multiple IManager instances can exist.
36
37 The following are the 'thread safety' requirements for a IJob:
38 o Data can safely be exchanged via Model Points
39 o The application can safely pass initial settings, configuration,
40 options when the IJob is started
41 */
42class IJob : public Kit::Container::KeyedItem, public Kit::Container::KeyLiteralString
43{
44public:
45 /** This method returns the IJob name. The IJob name must an printable
46 ASCII string with NO whitespace. The name is used to uniquely identify
47 (within an IManager instance) the individual IJob
48 */
49 virtual const char* getName() const noexcept = 0;
50
51 /** This method returns brief description/summary of what the IJob does.
52 Typically this information will be displayed on the Debug console to
53 aid the user on how to execute the IJob. The content/verbosity of the
54 text is IJob specific.
55 */
56 virtual const char* getDescription() const noexcept = 0;
57
58 /** This method returns help (optional) for the IJob's arguments.
59 Typically this information will be displayed on the Debug console to
60 aid the user on how to execute the IJob. The content/verbosity of the
61 text is IJob specific.
62 */
63 virtual const char* getUsage() const noexcept = 0;
64
65
66public:
67 /** This method has PACKAGE Scope, i.e. it is intended to be ONLY accessible
68 by other classes in the Kit::Job::IJob namespace. The Application should
69 NEVER call this method.
70
71 This method is used to perform any needed initialization on start-up of
72 Application. This method is called when the IJob's Manager is opened.
73
74 This method MUST be called in the thread that the IJob executes in.
75 */
76 virtual void initialize_() noexcept = 0;
77
78 /** This method has PACKAGE Scope, i.e. it is intended to be ONLY accessible
79 by other classes in the Kit::Job::IJob namespace. The Application should
80 NEVER call this method.
81
82 This method is used to perform any needed shutdown when the application
83 is shutdown. This method is called when the IJob's Manager is closed.
84
85 This method MUST be called in the thread that the IJob executes in.
86 */
87 virtual void shutdown_() noexcept = 0;
88
89
90public:
91 /** This method has PACKAGE Scope, i.e. it is intended to be ONLY accessible
92 by other classes in the Kit::Job::IJob namespace. The Application should
93 NEVER call this method, instead the Application must call the Manager
94 to start a IJob.
95
96 If the client has no 'optionalArgs' to pass to the IJob instance - it must
97 provide an empty/blank null terminated string, i.e can NOT pass a nullptr.
98
99 This method is used to start a IJob. If the IJob is unable to start then
100 false is returned; else true is returned. NOTE: The IJob is responsible
101 for setting its 'running state' to true when the start_() method returns
102 true.
103
104 This method MUST be called in the thread that the IJob executes in.
105
106 NOTE: The IJob is allowed to perform destructive parsing on 'optionalArgs',
107 however it must honor the string len of 'optionalArgs'
108 */
109 virtual bool start_( IContext& context, char* optionalArgs ) noexcept = 0;
110
111 /** This method has PACKAGE Scope, i.e. it is intended to be ONLY accessible
112 by other classes in the Kit::Job::IJob namespace. The Application should
113 NEVER call this method, instead the Application must call the Manager
114 to stop a IJob.
115
116 This method is used to stop a IJob. NOTE: The IJob is responsible for
117 setting its 'running state' to false when the stop_() method returns.
118 */
119 virtual void stop_() noexcept = 0;
120
121public:
122 /** This method has PACKAGE Scope, i.e. it is intended to be ONLY accessible
123 by other classes in the Kit::Job::IJob namespace. The Application should
124 NEVER call this method, instead the Application must call the Manager
125 to stop a IJob.
126
127 The method returns true if the IJob is currently running; else false.
128 */
129 virtual bool isRunning_() const noexcept = 0;
130
131protected:
132 /// Protected Constructor (needed because of the inheritance from KeyLiteralString)
133 IJob( const char* name ) noexcept
134 : Kit::Container::KeyLiteralString( name )
135 {
136 KIT_SYSTEM_ASSERT( name == nullptr || *name != '\0' );
137 }
138
139public:
140 /// Virtual destructor
141 virtual ~IJob() = default;
142};
