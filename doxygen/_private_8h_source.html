31/** Holds a lazily constructed, never destroyed instance of T.
32
33 Declare instances as function-local statics. Construction-on-first-use
34 removes the static initialization order dependency for globals that are
35 used from other translation units' static constructors.
36
37 This class MUST remain trivially destructible (see the static_assert
38 below) - a non-trivial destructor on a function-local static is registered
39 via atexit()/__cxa_atexit(), which pulls newlib's exit machinery (and the
40 undefined '_fini') into the bare-metal targets.
41 */
42template <class T>
43class NeverDestroyed
44{
45public:
46 ///
47 NeverDestroyed() noexcept
48 {
49 static_assert( std::is_trivially_destructible<NeverDestroyed>::value,
50 "NeverDestroyed<T> must stay trivially destructible - do NOT add a destructor" );
51 ::new ( m_storage ) T();
52 }
53
54 /// Returns the wrapped instance
55 T& get() noexcept { return *reinterpret_cast<T*>( m_storage ); }
56
57private:
58 alignas( T ) unsigned char m_storage[sizeof( T )];
59};
