32/** NV Record status values. Transitions are one-directional:
33 ERASED -> VALID -> INVALID. Each transition only requires clearing
34 bits (1->0), which flash hardware allows without an erase cycle.
35 */
36struct PageStatus_T
37{
38 static constexpr uint32_t ERASED = 0xFFFFFFFF; //!< Record slot is empty
39 static constexpr uint32_t VALID = 0xFFFF0000; //!< Record contains current data
40 static constexpr uint32_t INVALID = 0x00000000; //!< Record is obsolete
41};

44/** NV Record header structure (24 bytes). Stored at the beginning of each
45 NV Record in flash.
46 */
47struct PageHeader_T
48{
49 uint32_t magic; //!< Magic number (0xA5A5A5A5) identifying a valid header
50 uint32_t sequenceNum; //!< Monotonically increasing sequence number. Comparisons use
51 //!< 32-bit serial-number arithmetic so that wrap-around is handled
52 //!< (see Api::isSequenceNewer()).
53 uint32_t dataOffset; //!< Starting offset of this record within the logical NV space
54 uint32_t dataLength; //!< Length of the data payload (equals nvPageSize)
55 uint32_t crc32; //!< CRC32 over the preceding 16 bytes (magic through dataLength)
56 uint32_t status; //!< Record lifecycle state (ERASED, VALID, or INVALID)
57};

86/** This class implements the NV::IApi interface using a flash device (via
87 Flash::IApi) for persistent storage with wear leveling.
88
89 The implementation uses a log-structured approach where each write creates
90 a new NV Record containing a record header and a full page of data. A
91 read-modify-write pattern ensures that unmodified bytes within a page are
92 preserved.
93
94 An in-memory page map provides O(1) lookup from logical page index to
95 physical flash address. On startup, the page map is rebuilt by scanning
96 all NV Records in the flash region.
97
98 Sector reclamation uses a two-phase approach: first scan for already-erased
99 slots, and only erase a sector when no erased slots remain and the sector
100 contains no VALID records. To guarantee that reclamation can never
101 deadlock (a state where every sector holds a VALID record and none can be
102 erased), the flash region MUST be over-provisioned such that the number of
103 physical sectors is at least the number of logical pages
104 (numLogicalPages <= numSectors). start() enforces this invariant and fails
105 for any configuration that violates it.
106
107 Template parameter MAX_LOGICAL_PAGES defines the maximum number of logical
108 pages supported (determines size of the in-memory page map array).
109 */
110template <size_t MAX_LOGICAL_PAGES>
111class Api : public NV::IApi
112{
113protected:

123
124 The 'crcAlgo' MUST be a 32-bit wide CRC algorithm.
125 */

155 }

157 memset( m_pageMap, ERASED_BYTE_VALUE, sizeof( m_pageMap ) );

161public:
162 bool start( void* startArgs = nullptr ) noexcept override
163 {
164 if ( m_isStarted )
165 {
166 return true;
167 }
168
169 size_t sectorSize = m_flashDriver.getSectorSize();
170 size_t nvRegionSize = m_config.flashEndAddress - m_config.flashStartAddress;
171 m_numSectors = nvRegionSize / sectorSize;
172 m_pagesPerSector = sectorSize / getPhysicalPageSize();
173 m_totalPages = m_numSectors * m_pagesPerSector;
174
175 // Every erase issued by this class is flashStartAddress + N*sectorSize,
176 // so a misaligned base address would be rejected by the Flash driver at
177 // run time (ERR_ALIGNMENT). Fail at start() instead.
178 KIT_SYSTEM_ASSERT( m_config.flashStartAddress % sectorSize == 0 );
179 if ( m_config.flashStartAddress % sectorSize != 0 )
180 {
181 return false;
182 }
183
184 // Over-provisioning invariant. At most one record per logical page can
185 // be VALID at any one time, so when superseding logical page P there are
186 // at most (m_numLogicalPages - 1) VALID records belonging to *other*
187 // logical pages (P's current record is the one being invalidated). To
188 // deadlock sector reclamation, every sector would have to hold at least
189 // one such "other-page" VALID record, which requires
190 // (m_numLogicalPages - 1) >= m_numSectors, i.e. m_numLogicalPages >
191 // m_numSectors. Requiring m_numLogicalPages <= m_numSectors therefore
192 // guarantees findFreePageAddress() Phase 2 can always reclaim a sector.
193 // A configuration that violates this invariant is a design/configuration
194 // error, so refuse to start rather than risk a run-time write failure.
195 KIT_SYSTEM_ASSERT( m_numLogicalPages <= m_numSectors );
196 if ( m_numLogicalPages > m_numSectors )
197 {
198 return false;
199 }
200
201 memset( m_pageMap, ERASED_BYTE_VALUE, sizeof( m_pageMap ) );
202
203 if ( !scanAndBuildPageMap() )
204 {
205 return false;
206 }
207
208 m_isStarted = true;
209 return true;
210 }

212 void stop() noexcept override

214 m_isStarted = false;

218public:
219 bool write( size_t dstOffset,
220 const void* srcData,
221 size_t numBytesToWrite ) noexcept override
222 {
223 if ( !m_isStarted || srcData == nullptr || numBytesToWrite == 0 )
224 {
225 return false;
226 }
227
228 if ( dstOffset + numBytesToWrite > m_config.nvTotalSize )
229 {
230 return false;
231 }
232
233 const uint8_t* src = static_cast<const uint8_t*>( srcData );
234 size_t remaining = numBytesToWrite;
235 size_t offset = dstOffset;
236
237 while ( remaining > 0 )
238 {
239 size_t pageIndex = offsetToPageIndex( offset );
240 if ( pageIndex >= m_numLogicalPages )
241 {
242 return false;
243 }
244
245 size_t offsetInPage = offset % m_config.nvPageSize;
246 size_t bytesThisPage = m_config.nvPageSize - offsetInPage;
247 if ( bytesThisPage > remaining )
248 {
249 bytesThisPage = remaining;
250 }
251
252 if ( !readCurrentPageData( pageIndex, m_workBuffer ) )
253 {
254 return false;
255 }
256
257 memcpy( m_workBuffer + offsetInPage, src, bytesThisPage );
258
259 uint32_t oldPageAddr = m_pageMap[pageIndex];
260 uint32_t freeAddr = findFreePageAddress( oldPageAddr );
261 if ( freeAddr == INVALID_PAGE_ADDRESS )
262 {
263 return false;
264 }
265
266 PageHeader_T header;
267 header.magic = MAGIC_NUMBER;
268 header.sequenceNum = ++m_currentSequence;
269 header.dataOffset = static_cast<uint32_t>( pageIndex * m_config.nvPageSize );
270 header.dataLength = static_cast<uint32_t>( m_config.nvPageSize );
271 header.crc32 = calculateCrc( &header, CRC_HEADER_FIELD_SIZE );
272 header.status = PageStatus_T::VALID;
273
274 if ( m_flashDriver.write( freeAddr, &header, HEADER_SIZE ) != FlashResult_T::SUCCESS )
275 {
276 return false;
277 }
278
279 if ( m_flashDriver.write( freeAddr + HEADER_SIZE, m_workBuffer, m_config.nvPageSize ) != FlashResult_T::SUCCESS )
280 {
281 return false;
282 }
283
284 if ( m_pageMap[pageIndex] != INVALID_PAGE_ADDRESS )
285 {
286 if ( !markPageInvalid( m_pageMap[pageIndex] ) )
287 {
288 return false;
289 }
290 }
291
292 m_pageMap[pageIndex] = freeAddr;
293
294 src += bytesThisPage;
295 offset += bytesThisPage;
296 remaining -= bytesThisPage;
297 }
298
299 return true;
300 }

303 void* dstData,
304 size_t sizeDstData,
305 size_t numBytesToRead ) noexcept override
306 {
307 if ( !m_isStarted || dstData == nullptr || numBytesToRead == 0 )
308 {
309 return false;
310 }
311

318 {
319 return false;
320 }
321
322 uint8_t* dst = static_cast<uint8_t*>( dstData );
323 size_t remaining = numBytesToRead;
324 size_t offset = srcOffset;
325

327 {

329 if ( pageIndex >= m_numLogicalPages )
330 {
331 return false;
332 }
333
334 size_t offsetInPage = offset % m_config.nvPageSize;
335 size_t bytesThisPage = m_config.nvPageSize - offsetInPage;
336 if ( bytesThisPage > remaining )
337 {
338 bytesThisPage = remaining;
339 }
340
341 if ( m_pageMap[pageIndex] != INVALID_PAGE_ADDRESS )
342 {
343 size_t dataAddr = m_pageMap[pageIndex] + HEADER_SIZE + offsetInPage;
344 if ( m_flashDriver.read( dataAddr, dst, bytesThisPage ) != FlashResult_T::SUCCESS )
345 {
346 return false;
347 }
348 }
349 else
350 {
351 memset( dst, ERASED_BYTE_VALUE, bytesThisPage );
352 }
353
354 dst += bytesThisPage;
355 offset += bytesThisPage;
356 remaining -= bytesThisPage;
357 }
358
359 return true;

368 /// Erases all sectors and resets the page map, effectively formatting the NV storage.
369 bool format() noexcept
370 {
371 if ( !m_isStarted )
372 {
373 return false;
374 }
375
376 size_t sectorSize = m_flashDriver.getSectorSize();
377

379 {

381 if ( m_flashDriver.eraseSector( sectorAddr ) != FlashResult_T::SUCCESS )
382 {
383 return false;
384 }

394 /// Returns current statistics: cumulative erase count, free pages, and valid pages.
395 bool getStatistics( uint32_t& eraseCount,
396 size_t& freePages,
397 size_t& validPages ) noexcept
398 {
399 if ( !m_isStarted )
400 {
401 return false;
402 }
403
404 eraseCount = m_eraseCount;
405 freePages = 0;
406 validPages = 0;
407
408 size_t sectorSize = m_flashDriver.getSectorSize();
409 size_t physPageSize = getPhysicalPageSize();
410
411 for ( size_t i = 0; i < m_totalPages; i++ )
412 {
413 size_t sectorIndex = i / m_pagesPerSector;
414 size_t slotIndex = i % m_pagesPerSector;
415 size_t address = m_config.flashStartAddress + ( sectorIndex * sectorSize ) + ( slotIndex * physPageSize );
416
417 PageHeader_T header;
418 if ( m_flashDriver.read( address, &header, HEADER_SIZE ) != FlashResult_T::SUCCESS )
419 {
420 return false;
421 }
422
423 if ( header.magic == 0xFFFFFFFF && header.status == PageStatus_T::ERASED )
424 {
425 freePages++;
426 }
427 else if ( header.magic == MAGIC_NUMBER && header.status == PageStatus_T::VALID )
428 {
429 validPages++;
430 }
431 }
432
433 return true;
434 }

438 /// Scans all physical pages and rebuilds the logical-to-physical page map.
439 bool scanAndBuildPageMap() noexcept
440 {
441 size_t sectorSize = m_flashDriver.getSectorSize();
442 size_t physPageSize = getPhysicalPageSize();
443 uint32_t maxSeq = 0;
444 bool haveMaxSeq = false;
445
446 for ( size_t i = 0; i < m_totalPages; i++ )
447 {
448 size_t sectorIndex = i / m_pagesPerSector;
449 size_t slotIndex = i % m_pagesPerSector;
450 size_t address = m_config.flashStartAddress + ( sectorIndex * sectorSize ) + ( slotIndex * physPageSize );
451
452 PageHeader_T header;
453 if ( m_flashDriver.read( address, &header, HEADER_SIZE ) != FlashResult_T::SUCCESS )
454 {
455 return false;
456 }
457
458 if ( header.magic != MAGIC_NUMBER || header.status != PageStatus_T::VALID )
459 {
460 continue;
461 }
462
463 uint32_t expectedCrc = calculateCrc( &header, CRC_HEADER_FIELD_SIZE );
464 if ( header.crc32 != expectedCrc )
465 {
466 continue;
467 }
468
469 size_t logicalIndex = header.dataOffset / m_config.nvPageSize;
470 if ( logicalIndex >= m_numLogicalPages )
471 {
472 continue;
473 }
474
475 if ( !haveMaxSeq || isSequenceNewer( header.sequenceNum, maxSeq ) )
476 {
477 maxSeq = header.sequenceNum;
478 haveMaxSeq = true;
479 }
480
481 uint32_t currentPageAddress = m_pageMap[logicalIndex];
482 if ( currentPageAddress == INVALID_PAGE_ADDRESS )
483 {
484 m_pageMap[logicalIndex] = static_cast<uint32_t>( address );
485 }
486 else
487 {
488 PageHeader_T currentHeader;
489 if ( m_flashDriver.read( currentPageAddress, &currentHeader, HEADER_SIZE ) != FlashResult_T::SUCCESS )
490 {
491 return false;
492 }
493
494 if ( isSequenceNewer( header.sequenceNum, currentHeader.sequenceNum ) )
495 {
496 if ( !markPageInvalid( currentPageAddress ) )
497 {
498 return false;
499 }
500
501 m_pageMap[logicalIndex] = static_cast<uint32_t>( address );
502 }
503 else
504 {
505 if ( !markPageInvalid( static_cast<uint32_t>( address ) ) )
506 {
507 return false;
508 }
509 }
510 }
511 }
512
513 m_currentSequence = maxSeq;
514 return true;
515 }

517 /// Finds a free physical page slot, reclaiming a sector if necessary.
518 uint32_t findFreePageAddress( uint32_t aboutToInvalidate = INVALID_PAGE_ADDRESS ) noexcept
519 {
520 size_t sectorSize = m_flashDriver.getSectorSize();
521 size_t physPageSize = getPhysicalPageSize();
522
523 // Phase 1: Scan for an already-erased record slot
524 for ( size_t i = 0; i < m_totalPages; i++ )
525 {
526 size_t sectorIndex = i / m_pagesPerSector;
527 size_t slotIndex = i % m_pagesPerSector;
528 size_t address = m_config.flashStartAddress + ( sectorIndex * sectorSize ) + ( slotIndex * physPageSize );
529
530 PageHeader_T header;
531 if ( m_flashDriver.read( address, &header, HEADER_SIZE ) != FlashResult_T::SUCCESS )
532 {
533 continue;
534 }
535
536 if ( header.magic == PageStatus_T::ERASED )
537 {
538 return static_cast<uint32_t>( address );
539 }
540 }
541
542 // Phase 2: Try to reclaim a sector with no VALID records
543 for ( size_t s = 0; s < m_numSectors; s++ )
544 {
545 size_t sectorAddr = m_config.flashStartAddress + ( s * sectorSize );
546 bool hasValid = false;
547
548 for ( size_t slot = 0; slot < m_pagesPerSector; slot++ )
549 {
550 size_t slotAddr = sectorAddr + ( slot * physPageSize );
551
552 if ( static_cast<uint32_t>( slotAddr ) == aboutToInvalidate )
553 {
554 continue;
555 }
556
557 PageHeader_T header;
558 if ( m_flashDriver.read( slotAddr, &header, HEADER_SIZE ) != FlashResult_T::SUCCESS )
559 {
560 hasValid = true;
561 break;
562 }
563
564 if ( header.magic == MAGIC_NUMBER && header.status == PageStatus_T::VALID )
565 {
566 hasValid = true;
567 break;
568 }
569 }
570
571 if ( !hasValid )
572 {
573 if ( aboutToInvalidate != INVALID_PAGE_ADDRESS )
574 {
575 size_t aboutToInvalidateSector = aboutToInvalidate - ( aboutToInvalidate % sectorSize );
576 if ( aboutToInvalidateSector == sectorAddr )
577 {
578 markPageInvalid( aboutToInvalidate );
579 }
580 }
581
582 if ( m_flashDriver.eraseSector( sectorAddr ) == FlashResult_T::SUCCESS )
583 {
584 m_eraseCount++;
585 return static_cast<uint32_t>( sectorAddr );
586 }
587 }
588 }
589
590 return INVALID_PAGE_ADDRESS;
591 }

636 /** Returns true if sequence number 'a' is newer than sequence number 'b',

638 arithmetic (RFC 1982 style): the signed difference is positive when 'a'
639 is ahead of 'b' within a half-range window, so a freshly wrapped small
640 value (e.g. 1) is correctly treated as newer than a pre-wrap large value
641 (e.g. 0xFFFFFFFF).
642 */
