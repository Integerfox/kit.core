75/** This class provides a collection of formatting method(s) for log entries
76 */
77class Formatter
78{
79public:
80 /** Selects how the time portion of an entry's timestamp (see
81 Kit::Time::getBootTime()) is rendered. The timestamp's time portion is
82 elapsed time since power-up unless the application has configured boot
83 time to use absolute time (i.e. USE_KIT_TIME_BOOTTIME_WITH_ABSOLUTETIME)
84 */
85 enum class TimestampFormat_T
86 {
87 eABSOLUTE, //!< UTC date/time, i.e. the time is milliseconds since the Epoch: YYYY-MM-DD HH:MM:SS.mmm
88 eELAPSED //!< Elapsed time, i.e. the time is milliseconds since power-up: DD HH:MM:SS.mmm
89 };
90
91 /** The default timestamp format, i.e. the format that matches the time
92 source of Kit::Time::getBootTime(). USE_KIT_TIME_BOOTTIME_WITH_ABSOLUTETIME
93 (defined in kit_config.h) is the source of truth: eABSOLUTE when it is
94 defined, else eELAPSED.
95
96 NOTE: An application that provides its own getBootTime() (see
97 USE_KIT_TIME_CUSTOM_GET_BOOT_TIME) that returns absolute time must
98 also define USE_KIT_TIME_BOOTTIME_WITH_ABSOLUTETIME.
99 */
100 static constexpr TimestampFormat_T defaultTimestampFormat = TimestampFormat_T::KIT_LOGGING_FORMATTER_DEFAULT_TIMESTAMP_FORMAT_;
101
102public:
103 /** Converts a binary EntryData_T struct instance to formatted text. The
104 application is responsible for ensuring the 'dstStringBuf' argument
105 can contain the maximum formatted size (the default size is
106 OPTION_KIT_LOGGING_FORMATTER_MAX_TEXT_LEN). The 'persistentStorageId'
107 argument is optional and represents the persistent storage ID for the
108 entry being converted. The 'timestampFormat' argument selects how the
109 timestamp is rendered; it defaults to defaultTimestampFormat. The
110 format shown below is the eABSOLUTE format; eELAPSED replaces
111 YYYY-MM-DD HH:MM:SS.mmm with DD HH:MM:SS.mmm.
112
113 The method returns true when successful. False is returned if the provided
114 string buffer was not large enough to hold the formatted entry.
115
116 \code
117
118 The format is:
119 [<storageIdAsciiHex>] (<bootCnt>:YYYY-MM-DD HH:MM:SS.mmm) <appState><classification>-<packageId>-<subSystemId>-<msgId>: <messageText>
120 OR
121 (<bootCnt>:YYYY-MM-DD HH:MM:SS.mmm) <appState><classification>-<packageId>-<subSystemId>-<msgId>: <messageText>
122
123 Where <appState> is the Application's rendering of the entry's application
124 state (see IApplication::applicationStateToString()). By default this is
125 an empty string. The rendering is truncated to
126 OPTION_KIT_LOGGING_FORMATTER_APPLICATION_STATE_TEXT_LEN characters.
127 \endcode
128 */
129 static bool toString( IApplication& application,
130 const Kit::Logging::Framework::EntryData_T& srcEntryToFormat,
131 Kit::Text::IString& dstStringBuf,
132 uint64_t persistentStorageId = UINT64_MAX,
133 TimestampFormat_T timestampFormat = defaultTimestampFormat ) noexcept;
134
135 /// Helper method: Appends the formatted timestamp to 'dstStringBuf' (uses the same timestamp formatting as toString())
136 static void appendFormattedTimestamp( uint64_t timestamp,
137 Kit::Text::IString& dstStringBuf,
138 TimestampFormat_T timestampFormat = defaultTimestampFormat ) noexcept;
139};
