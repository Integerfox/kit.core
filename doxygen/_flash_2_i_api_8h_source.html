27/** This class defines the interface for a platform independent Flash memory
28 driver. The interface provides read, write, and erase operations. Erasing
29 is performed at sector granularity (one or more sectors) or the full chip.
30
31 The interface assumes a flash device organized into sectors (smallest
32 erasable unit) and pages (largest single-program unit). Writes that
33 cross page boundaries are handled internally by the implementation.
34
35 The interface itself is NOT thread safe. It is the responsibility of
36 the users/clients of the driver to handle any threading issues.
37 */
38class IApi : public virtual IStart,
39 public virtual IStop
40{
41public:
42 /// Return codes for flash operations
43 enum Result_T
44 {
45 SUCCESS = 0, //!< Operation completed successfully
46 ERR_RANGE, //!< Operation failed due to out-of-range access, e.g. address + numBytes > getTotalSize()
47 ERR_ALIGNMENT, //!< Operation failed due to improper address/sector alignment
48 ERR_FAILED //!< Operation failed (generic error)
49 };
50
51public:
52 /** This method reads 'numBytes' from flash starting at 'srcOffset'
53 into 'dstBuffer'. The application is responsible for ensuring that
54 'dstBuffer' is at least 'numBytes' in length.
55
56 The method returns Result_T::SUCCESS if the operation was successful;
57 else an error code is returned.
58 */
59 virtual Result_T read( size_t srcOffset,
60 void* dstBuffer,
61 size_t numBytes ) noexcept = 0;
62
63 /** This method writes 'numBytes' from 'srcBuffer' to flash starting
64 at 'dstOffset'. Writes that cross flash page boundaries are handled
65 automatically. The application is responsible for ensuring that
66 'srcBuffer' is at least 'numBytes' in length.
67
68 The method returns Result_T::SUCCESS if the operation was successful;
69 else an error code is returned.
70 */
71 virtual Result_T write( size_t dstOffset,
72 const void* srcBuffer,
73 size_t numBytes ) noexcept = 0;
74
75 /** This method erases 'numberOfSectors' flash sectors (typically 4KB each)
76 starting at the specified sector-aligned address.
77
78 Passing a 'numberOfSectors' greater than one allows the concrete
79 implementation to optimize the request (e.g. converting eight 4KB
80 sector erases into a single 32KB block erase command).
81
82 The method returns Result_T::SUCCESS if the operation was successful;
83 else an error code is returned.
84 */
85 virtual Result_T eraseSector( size_t sectorAddress,
86 unsigned numberOfSectors = 1 ) noexcept = 0;
87
88 /** This method erases the entire flash chip.
89
90 The method returns Result_T::SUCCESS if the operation was successful;
91 else an error code is returned.
92 */
93 virtual Result_T eraseChip() noexcept = 0;
94
95 /** This method blocks until the Flash is 'ready' (i.e. not busy with a
96 physical write or erase operation). The method returns true when the
97 flash is ready; else false is returned if the timeout period expired
98 before the flash was ready.
99 */
100 virtual bool waitUntilReady( uint32_t timeoutMs = 1000 ) noexcept = 0;
101
102
103public:
104 /** Returns the total flash device size in bytes */
105 virtual size_t getTotalSize() const noexcept = 0;
106
107 /** Returns the flash sector size in bytes (smallest erasable unit) */
108 virtual size_t getSectorSize() const noexcept = 0;
109
110 /** Returns the flash write page size in bytes (largest single-program
111 unit used by the write() method)
112 */
113 virtual size_t getWritePageSize() const noexcept = 0;
114
115 /** Returns the total number of sectors in the flash device */
116 virtual size_t getNumSectors() const noexcept = 0;
117
118
119public:
120 /// Virtual destructor
121 virtual ~IApi() noexcept = default;
122};
