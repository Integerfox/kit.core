37 */

47 : ServerBase( timeOutPeriodInMsec, eventFlagsList, watchdog )

49 }

51public:
52 /// See Kit::EventQueue::IChangeNotification

56 }
57
58 /// See Kit::EventQueue::IChangeNotification

61 m_changeHandler.removePendingChangingNotification( observer );

63
64protected:
65 /// See Kit::EventQueue::ServerBase
