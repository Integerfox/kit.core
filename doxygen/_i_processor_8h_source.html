23/** This abstract class defines the interface a TShell Command Processor. The
24 Command Processor is responsible for scanning the input, determining
25 what command (if any) to execute; and then executing the command.
26
27 The design of the interface supports running the Command Processor with
28 blocking-thread and with cooperative scheduling semantics.
29*/
30class IProcessor
31{
32public:
33 /** This method is used to start the Command Processor, i.e. it will
34 begin to process commands.
35
36 When 'blocking' is set to true, the method command will not return until
37 the Command Processor self terminates or a Input/Output stream error was
38 encounter. The method returns true if the Command Processor self
39 terminated or was requested to stop; else false is returned e.g. a
40 Input/Output stream error was encounter).
41
42 When 'blocking' is set to false, the command processor will be started
43 and the method returns immediately. After start() is the called the
44 Application is responsible for calling poll() to provide the command
45 processor with CPU time to process commands. The method returns true if
46 the Command Processor was successfully started; else false is returned
47 (e.g. a Input/Output stream error was encounter).
48
49 NOTE: This method is an 'in-thread' initialization, i.e. not thread safe.
50 The application is RESPONSIBLE for managing threading issues.
51 */
52 virtual bool start( Kit::Io::IInput& infd,
53 Kit::Io::IOutput& outfd,
54 bool blocking = true ) noexcept = 0;
55
56
57 /** This method is used to provide the command processor 'CPU cycles' to
58 parse/process/execute commands. This command should ONLY be called when
59 the application called the start() method with the 'blocking' argument
60 set to false. This method should be called as often as possible, e.g.
61 every pass of the 'main loop'
62
63 NOTE: This method must ALWAYS be called from the same 'thread context'
64 that called the start() method (i.e. not thread safe).
65
66 The method returns 0 if the method executed without errors. The method
67 returns -1 if an error occurred (e.g. a Input/Output stream error was
68 encounter). The method returns 1 if the command processor self terminated,
69 not started, or was requested to stop.
70 */
71 virtual int poll() noexcept = 0;
72
73 /** This non-blocking method requests the Command Processor to stop. When
74 (or if) the Command Processor actually stops depends on the target's
75 implementation, health of the Shell, current command(s) executing, etc.
76 This method returns immediately. There is no feedback/confirmation
77 when the Command Processor stops.
78
79 This command IS thread safe and can be called from any thread context.
80 */
81 virtual void requestStop() noexcept = 0;
82
83public:
84 /// Virtual destructor
85 virtual ~IProcessor() = default;
86};
