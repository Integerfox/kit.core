23/** Data structure used to exchange data between a Client and Server for the
24 'Foo' transaction.
25
26 The following are needed because the structure is used as a 'data source'
27 for a Model Point:
28 ----------------------------------------------------------------------------
29 - The comparison operation
30 - The copy constructor and copy operator are implemented to ensure that
31 any 'pad' bytes in the structure get zero'd out. This is desired because
32 the default 'comparison operation' in the DM framework uses memcmp. A
33 more sophisticated comparison approach can be used - but it requires the
34 concrete Model Point to supply/implement the comparison method.
35 */
36struct Foo_T
37{
38 int32_t upperLimit; //!< IN: Upper limit for the random number generator
39 int32_t lowerLimit; //!< IN: Lower limit for the random number generator
40 int32_t randomNumber; //!< OUT: Random number generated by the Server and sent to the Client
41 bool isValid; //!< OUT: Indicates whether the 'randomNumber' value is valid (true) or not (false)
42
43
44 /// Constructor (to ensure any pad bytes get zero'd -->needed for brute force compare operation)
45 Foo_T()
46 {
47 memset( (void*)this, 0, sizeof( Foo_T ) );
48 }
49
50 /// Copy Constructor (to ensure any pad bytes get zero'd)
51 Foo_T( const Foo_T& other )
52 {
53 memcpy( (void*)this, (void*)&other, sizeof( Foo_T ) );
54 }
55
56 /// Constructor - all arguments
57 Foo_T( int32_t upperLimit,
