94/** Data stored by the RunControl Model Point.
95
96 The following are needed because the structure is used as a 'data source'
97 for a Model Point:
98 ----------------------------------------------------------------------------
99 - The comparison operation
100 - The constructors, copy constructor, and copy operator are implemented to
101 ensure that any 'pad' bytes in the structure get zero'd out. This is
102 desired because the default 'comparison operation' in the DM framework
103 uses memcmp.
104 */
105struct RunControl_T
106{
107 /** Identifier of the most recent request. The identifier is incremented
108 for each new request (and never zero once a request has been made), i.e.
109 it allows a requestor to correlate a result with its request even when
110 change notifications have been coalesced.
111 */
112 uint16_t requestId;
113
114 /// The most recent requested state
115 RunRequest requested;
116
117 /// The actual state of the widget
118 RunState actual;
119
120 /// Status of the most recent request
121 RunRequestStatus status;
122
123
124 /// Constructor (to ensure any pad bytes get zero'd -->needed for brute force compare operation)
125 RunControl_T()
126 {
127 memset( (void*)this, 0, sizeof( RunControl_T ) );
128 }
129
130 /// Copy Constructor (to ensure any pad bytes get zero'd)
131 RunControl_T( const RunControl_T& other )
132 {
133 memcpy( (void*)this, (void*)&other, sizeof( RunControl_T ) );
134 }
135
136 /// Constructor - all arguments
137 RunControl_T( uint16_t requestId,
138 RunRequest requested,
139 RunState actual,
140 RunRequestStatus status )
141 {
142 memset( (void*)this, 0, sizeof( RunControl_T ) );
143 this->requestId = requestId;
144 this->requested = requested;
145 this->actual = actual;
146 this->status = status;
147 }
148
149 /// Comparison operator (for DM support)
150 bool operator==( RunControl_T const& other ) const
151 {
152 return memcmp( this, &other, sizeof( RunControl_T ) ) == 0;
153 }
154
155 /// Copy operator
156 RunControl_T& operator=( const RunControl_T& other )
157 {
158 memcpy( (void*)this, (void*)&other, sizeof( RunControl_T ) );
159 return *this;
160 }
161};

164/** This class provides a Model Point that is the 'data' for a closed-loop
165 request/response interface to a 'runnable' widget, i.e. a widget that can be
166 started, stopped, paused, and resumed. The Model Point only holds the data
167 (and enforces the data's consistency); the policies for using the data are
168 implemented by the RunRequestor and RunManager 'pseudo' Model Points. The
169 RunRequestor and RunManager are the ONLY intended interfaces for modifying
170 the data.
171
172 The concrete Model Point is invalid until the RunManager publishes the
173 widget's actual state. A request is not accepted while the Model Point is
174 invalid.
175
176 The actual state ERRORED is a terminal state, i.e. once in the ERRORED
177 state, the state cannot be changed (short of invalidating the Model Point)
178 and no new requests are accepted.
179
180 IMPORTANT: The Model Point's data is transient/run-time state and MUST NEVER
181 be persistently stored.
182
183 The toJSON() format is:
184 \code
185
186 { name:"<mpname>", type:"<mptypestring>", valid:true|false, seqnum:nnnn, locked:true|false,
187 val:{requested:"<RunRequest>", actual:"<RunState>", status:"<RunRequestStatus>", requestId:nnnn}
188 }
189
190 \endcode
191
192 The fromJSON() is NOT supported, i.e. the Model Point cannot be modified via
193 a JSON object (use the pseudo Model Points instead).
194
195 NOTE: All methods in this class ARE thread Safe unless explicitly
196 documented otherwise.
197 */
