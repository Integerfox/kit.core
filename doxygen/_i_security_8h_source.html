38/** This abstract class defines the interface validating that is okay to execute
39 a TShell command. The underlying security model is specific to the application
40 and is not defined by this interface.
41*/
42class ISecurity
43{
44public:
45 /** This method evaluates the current security state and returns true/false
46 if the specified permission level is sufficient to execute a command.
47
48 The 'queryOnly' parameter is used to indicate that the caller is only
49 interested in querying the security state and not actually executing
50 the command. A use case for this is the TShell 'help' command that
51 only displays authorized commands but does not execute them.
52 */
53 virtual bool isAuthorized( Permissions_T permission,
54 const char* entireCommandString,
55 bool queryOnly = false ) noexcept = 0;
56
57
58 /** This method is used to suppress the TShell outputs until the session
59 has been 'authenticated'. This can be useful in scenarios where you
60 want to prevent unauthorized users from seeing certain output until they
61 have successfully authenticated. The outputs that are suppressed - when
62 the method returns true - include, but are not limited to, the
63 following:
64
65 - The TShell prompt
66 - The TShell welcome and farewell messages
67 - The TShell command error messages (e.g. unrecognized command, bad
68 syntax, etc.)
69
70 NOTE: Output generated by individual commands (i.e. via the IContext's
71 writeFrame() method) is NOT automatically suppressed by this method.
72 It is the responsibility of individual commands to check the
73 security state (via isSilent()) and suppress their output as needed.
74 */
75 virtual bool isSilent() noexcept = 0;
76
77public:
78 /// Virtual destructor
79 virtual ~ISecurity() = default;
80};
