24/** This class implements a brute force approach to initializing a 'pico
25 semaphore' when the Kit C++ library is initialized. This is to support
26 statically allocated/created Kit::System::BSemaphore instances. The issue
27 here is that 'pico semaphore' need to be initialized AFTER main() is called.
28
29 Note: The SDK semaphore is created with a max permit count of one - this is
30 what provides the binary (i.e. saturate at one) semantics.
31 */
32class BSemaphore_T : public Kit::System::IStartupHook
33{
34public:
35 /** Constructor.
36 Note: The registration is assumed to be done from a static allocated
37 instance (i.e. before main() and Kit::System::Api::initialize() is
38 called). However, registering AFTER wards is a don't care because
39 the 'start-up list' is only used during the call to Kit::System::Api::initialize()
40 */
41 BSemaphore_T( bool initialStateIsOne )
42 : IStartupHook( SYSTEM )
43 , m_initialStateIsOne( initialStateIsOne )
44 {
45 }
46
47
48protected:
49 /// This method is called when Kit::System::initialize() executes
50 void notify( InitLevel init_level ) noexcept override
51 {
52 sem_init( &m_sdkSema, m_initialStateIsOne ? 1 : 0, 1 );
53 }
54
55public:
