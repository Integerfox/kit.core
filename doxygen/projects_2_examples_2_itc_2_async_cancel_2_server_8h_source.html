40/** This concrete class is a 'server' that produces the LED flash patterns. The
41 server accepts flash requests from many clients. The server responses to
42 a client AFTER it has completed the client's requested duty cycles. Flash
43 requests are queued in FIFO order. A software timer is used to manage the
44 LED flashing.
45 */
46class Server : public Kit::Itc::OpenCloseSync, public IFlashRequest, public Kit::System::Timer
47{

59 /// See Kit::Itc::OpenCloseSync
60 void request( OpenMsg& msg ) noexcept override
61 {
62 // Only 'open' when not already opened
63 if ( !m_opened )
64 {
65 // Housekeeping
66 m_opened = true;
67 m_currentFlashMsg = nullptr;
68 m_onCycle = false;

76 /// See Kit::Itc::OpenCloseSync
77 void request( CloseMsg& msg ) noexcept override
78 {
79 // Only 'close' when already opened
80 if ( m_opened )
81 {
82 // Housekeeping
83 m_opened = false;
84 Timer::stop();
85
86 // Return the current flash message to the sender if it exists
87 if ( m_currentFlashMsg != nullptr )
88 {
89 m_currentFlashMsg->getPayload().success = false;
90 m_currentFlashMsg->returnToSender();
91 m_currentFlashMsg = nullptr;
92 }
93
94 // Drain the request queue
95 auto* msg = m_flashRequestList.get();
96 while ( msg != nullptr )
97 {
98 msg->getPayload().success = false;
99 msg->returnToSender();
100 msg = m_flashRequestList.get();
101 }
102 }
103
104 // Return the message to the sender since we are done processing the request
105 msg.returnToSender();
106 }

110 /// See IFlashRequest
111 void request( FlashMsg& msg ) noexcept override
112 {
113 // NOTE: Make sure to have a *reference* to the payload (instead of a copy)
114 // since we need to update the 'success' field in the payload to
115 // indicate pass/fail of the request
116 auto& payload = msg.getPayload();
117
118 // Validate the flash rate request
119 if ( ( payload.flashPattern.onTimeMs < OPTION_EXAMPLES_ITC_ASYNCCANCEL_SERVER_MIN_FLASH_RATE_MS ) ||
120 ( payload.flashPattern.onTimeMs > OPTION_EXAMPLES_ITC_ASYNCCANCEL_SERVER_MAX_FLASH_RATE_MS ) ||
121 ( payload.flashPattern.offTimeMs < OPTION_EXAMPLES_ITC_ASYNCCANCEL_SERVER_MIN_FLASH_RATE_MS ) ||
122 ( payload.flashPattern.offTimeMs > OPTION_EXAMPLES_ITC_ASYNCCANCEL_SERVER_MAX_FLASH_RATE_MS ) ||
123 ( payload.flashPattern.onTimeMs == 0 && payload.flashPattern.offTimeMs == 0 ) ||
124 ( payload.flashPattern.repeatCount == 0 ) )
125 {
126 // Invalid flash rate request. Set the response value to false to indicate failure and return
127 payload.success = false;
128 msg.returnToSender(); // Respond to the client immediately
129 return;
130 }
131
132 // Queue the request if currently processing another flash request
133 if ( m_currentFlashMsg != nullptr )
134 {
135 m_flashRequestList.put( msg );
136 return;
137 }
138 m_currentFlashMsg = &msg;
139
140 // Begin the duty cycle(s)
141 // NOTE: The client's message is returned AFTER the duty cycle(s) have completed
142 beginDutyCycleSequence();
143 }

145 /// See IFlashRequest
146 void request( FlashCancelMsg& msg ) noexcept override
147 {
148 KIT_SYSTEM_TRACE_MSG( "main", "Received flash cancel request. %p, %p", m_currentFlashMsg, msg.getPayload().flashMsgToCancel );
149
150 // Default the results to: failed/not-cancelled
151 auto& payload = msg.getPayload();
152 payload.canceled = false;
153
154 // Is the request in progress
155 if ( m_currentFlashMsg != nullptr && m_currentFlashMsg == payload.flashMsgToCancel )
156 {
157 KIT_SYSTEM_TRACE_MSG( "main", "Cancelling current request..." );
158 Timer::stop();
159 Bsp_turn_off_debug1();
160 payload.canceled = true;
161 m_currentFlashMsg->getPayload().success = false; // Mark the transaction as failed due to cancellation
162 m_currentFlashMsg->returnToSender();
163 startNextRequest();
164 }
165
166 // Search the request queue for the message being canceled
167 else
168 {
169 auto* queuedMsgPtr = m_flashRequestList.first();
170 while ( queuedMsgPtr != nullptr )
171 {
172 if ( queuedMsgPtr == payload.flashMsgToCancel )
173 {
174 KIT_SYSTEM_TRACE_MSG( "main", "Cancelling queued request..." );
175 m_flashRequestList.remove( *queuedMsgPtr );
176 payload.canceled = true;
177 queuedMsgPtr->getPayload().success = false; // Mark the queued transaction as failed due to cancellation
178 queuedMsgPtr->returnToSender();
179 break; // Exit the loop once the Requestmessage is found and removed
180 }
181 queuedMsgPtr = m_flashRequestList.next( *queuedMsgPtr ); // Move to the next message in the queue
182 }
183 }
184
185 // Return the 'Cancel' message (NOTE: Must be returned AFTER returning the canceled request message)
186 msg.returnToSender();
187 }

189protected:
190 /// See Kit::System::ICounter, i.e. software timer expired callback
191 void expired() noexcept override
192 {
193 // ON Cycle in progress
194 if ( m_onCycle )
195 {
196 m_onCycle = false;
197 Bsp_turn_off_debug1();
198 Timer::start( m_currentFlashMsg->getPayload().flashPattern.offTimeMs );
199 }
200
201 // OFF Cycle in progress
202 else
203 {
204 m_repeatCount++;
205 m_onCycle = true;
206 KIT_SYSTEM_TRACE_MSG( "main", "LED off (%" PRIu32 " ms, cycleCount=%" PRIu32 ")", m_currentFlashMsg->getPayload().flashPattern.offTimeMs, m_repeatCount );
207
208 if ( m_repeatCount >= m_currentFlashMsg->getPayload().flashPattern.repeatCount )
209 {
210 m_currentFlashMsg->getPayload().success = true;
211 m_currentFlashMsg->returnToSender();
212 startNextRequest();
213 }
214 else
215 {
216 startOnCycle();
217 }
218 }
219 }
220
221protected:
222 /// Helper
223 void beginDutyCycleSequence() noexcept
224 {
225 m_repeatCount = 0;
226 startOnCycle();
227 }
228
229 /// Helper
230 void startOnCycle() noexcept
231 {
232 m_onCycle = true;
233 Bsp_turn_on_debug1();
234 auto onTime = m_currentFlashMsg->getPayload().flashPattern.onTimeMs;
235 KIT_SYSTEM_TRACE_MSG( "main", "LED ON (%" PRIu32 " ms, cycleCount=%" PRIu32 ")", onTime, m_repeatCount + 1 );
236 Timer::start( onTime );
237 }
238
239 /// Helper:
240 void startNextRequest() noexcept
241 {
242 m_currentFlashMsg = m_flashRequestList.get();
243 if ( m_currentFlashMsg != nullptr )
244 {
245 beginDutyCycleSequence();
246 }
247 }
