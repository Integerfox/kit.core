23/** This template class implements a message that a component posts to its OWN
24 event queue in order to yield execution back to its event loop and then be
25 called back as soon as possible - without busy-waiting and without
26 recursing on its own call stack.
27
28 THIS IS DELIBERATELY *NOT* THE ITC CLIENT/SERVER REQUEST PATTERN. There is
29 no SAP, no payload, no 'request()' method and no ReturnHandler because
30 there is no second thread, nothing to transfer and nothing to return to.
31 The sender and the receiver are the same component in the same thread; the
32 event queue is being used purely as a deferred-execution queue. Use
33 Kit::Itc::RequestMessage (or the SAP/OpenApi/CloseApi helpers) for genuine
34 cross-thread client/server requests.
35
36 The typical use case is a component that performs a long running sequence of
37 steps in the thread that owns its event queue. Instead of running every step
38 in a single callback - which starves the thread's other messages, event
39 flags and timers - the component performs one step, calls schedule(), and
40 returns.
41
42 Mechanics:
43 o schedule() posts this message to the component's own event queue,
44 i.e. the message is queued at the TAIL of the mailbox FIFO.
45 o Kit::Itc::Mailbox::processMessages() dispatches at MOST one message
46 per pass of the event loop, so the yield is fair with respect to
47 other ITC traffic directed at the thread.
48 o Because a message is pending, Kit::System::EventLoop skips its
49 semaphore wait, so the callback occurs on the very next pass of the
50 event loop instead of waiting for the thread's timing tick.
51 o Event flags and software timers are serviced on that same pass, so
52 neither is starved by a chain of continuations.
53
54 Compared to Kit::System::Timer::start(0) - which also defers the callback,
55 but only re-fires on the next timer tick - this mechanism resumes
56 immediately. Use a timer when the intent is to pace/throttle the work; use
57 this class when the intent is purely to yield.
58
59 THREADING INVARIANT: schedule(), cancel() and isPending() MUST be called
60 ONLY from the thread that owns 'myQueue'. Unlike ordinary cross-thread
61 ITC use, the component retains ownership of this message while it is queued.
62 This is safe ONLY because the sender and receiver are the same thread, so
63 the thread itself serializes all access to m_pending. Calling schedule()
64 from any other thread races on m_pending and can double-post the message,
65 violating the intrusive-container single-membership rule and causing a
66 fatal error.
67
68 LIFETIME: The message memory is owned by the component, and a scheduled
69 message is referenced by the mailbox FIFO. The component must therefore
70 outlive any scheduled callback; in particular, it must not be destroyed
71 while this message is queued.
72
73 BOUNDED CHAINS: While a message is pending the event loop does not block on
74 its semaphore, so a continuation chain spins the thread at 100%. The chain
75 MUST terminate, i.e. the component must eventually stop calling schedule().
76 Use a timer for open ended polling.
77
78 Template args:
79 CONTEXT Type of the class that implements the context for the
80 callback.
81
82 Example:
83 @code
84 #include "Kit/Itc/OpenCloseSync.h"
85 #include "Kit/Itc/SelfPostMessage.h"
86
87 class MyComponent : public Kit::Itc::OpenCloseSync
88 {
89 public:
90 MyComponent( Kit::EventQueue::IQueue& myQueue )
91 : Kit::Itc::OpenCloseSync( myQueue )
92 , m_resume( myQueue, *this, &MyComponent::nextStep )
93 , m_step( 0 )
94 , m_opened( false )
95 {
96 }
97
98 void request( OpenMsg& msg ) noexcept override
99 {
100 if ( !m_opened )
101 {
102 m_opened = true;
103 m_step = 0;
104 m_resume.schedule(); // Will run on the next Event Loop iteration but AFTER the OpenMsg has been returned
105 }
106 msg.getPayload().success = true;
107 msg.returnToSender();
108 }
109
110 void request( CloseMsg& msg ) noexcept override
111 {
112 if ( m_opened )
113 {
114 m_opened = false;
115 m_resume.cancel();
116 }
117 msg.getPayload().success = true;
118 msg.returnToSender();
119 }
120
121 protected:
122 void nextStep() noexcept
123 {
124 doStep( m_step++ );
125 if ( m_step < NUM_STEPS )
126 {
127 m_resume.schedule(); // yield, then resume
128 }
129 }
130
131 Kit::Itc::SelfPostMessage<MyComponent> m_resume;
132 unsigned m_step;
133 bool m_opened;
134 };
135 @endcode
136 */

146public:
147 /// Definition of the call-back method that resumes the component after the yield
148 typedef void ( CONTEXT::*ResumeFunction_T )();
149
150public:
151 /// Constructor. 'myQueue' MUST be the event queue of the owning thread
152 SelfPostMessage( Kit::EventQueue::IQueue& myQueue,
153 CONTEXT& context,
154 ResumeFunction_T resumeCallbackFunc ) noexcept
155 : m_queue( myQueue )
156 , m_context( context )
157 , m_callbackFuncPtr( resumeCallbackFunc )
158 , m_pending( false )
159 , m_cancelled( false )
160 {
161 }
162
163public:
164 /** Yields the processor back to the event loop and guarantees that the
165 callback method is invoked on a subsequent pass of that loop.
166
167 Calling schedule() more than once before the callback runs coalesces
168 into a SINGLE callback - the message is only posted when it is not
169 already in the mailbox. This makes the call safe to issue
170 speculatively, and it prevents a double-post that would violate the
171 intrusive-container single-membership rule.
172
173 Calling schedule() after cancel() - but before the cancelled message
174 has drained - re-arms the already-queued message.
175 */
176 void schedule() noexcept
177 {
178 m_cancelled = false;
179 if ( !m_pending )
180 {
181 m_pending = true;
182 m_queue.post( *this );
183 }
184 }
185
186 /** Cancels a pending callback, i.e. the callback method will NOT be
187 called. It is okay to call cancel() when nothing is scheduled.
188
189 Note: A posted message cannot be pulled back out of the mailbox FIFO,
190 so the message is still dispatched - it simply does nothing.
191 isPending() therefore stays true until the event loop drains it.
192 */
193 void cancel() noexcept
194 {
195 m_cancelled = m_pending;
196 }
197
198 /// Returns true when the message is in the mailbox FIFO, i.e. not yet dispatched
199 bool isPending() const noexcept
200 {
201 return m_pending;
202 }
203
204public:
205 /// See Kit::Itc::IMessage
206 void process() noexcept override
207 {
208 // Clear BEFORE the callback so that the callback can re-schedule me
209 m_pending = false;
210 if ( m_cancelled )
211 {
212 m_cancelled = false;
213 return;
214 }
215
216 ( m_context.*m_callbackFuncPtr )();
217 }
218
219protected:
220 /// The event queue I post myself to
221 Kit::EventQueue::IQueue& m_queue;
222
223 /// The component that owns me
224 CONTEXT& m_context;
225
226 /// The method that resumes the component
227 ResumeFunction_T m_callbackFuncPtr;
228
229 /// Set while I am in the mailbox FIFO
230 bool m_pending;
231
232 /// Set when a pending callback has been cancelled
233 bool m_cancelled;
234};
