28/** This partial concrete class provides common implementations, utility methods,
29 etc. for concrete child classes that implement the IChunk interface. The
30 class ASSUMES the following:
31
32 - All child classes will use a checksum to detect corrupted data.
33
34 - There is sufficiently large temporary buffer to hold a Record's data plus
35 ALL of the chunk's meta data (i.e. the entire chunk is assembled in the
36 buffer so it can be transferred with a single IMedia call). Note: The private_.h header
37 file provides a default buffer. See private_.h for additional details
38 about the default buffer.
39
40 - Some of helper methods (e.g. pushToRecord() and pullFromRecord()) are
41 virtual wrapper methods. By having a virtual wrapper - pushing/pulling
42 to/from the data record facilitates future child classes to extend existing
43 chunk classes with the ability to decrypt/encrypt the record data
44 */
45class Common : public IChunk
46{
47protected:
48 /// Constructor
49 Common( IMedia& media,
50 Kit::Checksum::IEdc& edc,
51 uint8_t* workBuffer,
52 Size_T workBufferSize ) noexcept
53 : m_media( media )
54 , m_crc( edc )
55 , m_workBuffer( workBuffer )
56 , m_workBufferSize( workBufferSize )
57 {
58 KIT_SYSTEM_ASSERT( m_workBuffer != nullptr && m_workBufferSize > 0 );
59 }
60
61protected:
62 /** Helper method that reads an entire chunk (header + record data + EDC)
63 from 'media' at 'index' into m_workBuffer and validates its EDC. The
64 chunk layout in m_workBuffer is:
65
66 [header: 'headerLen' bytes][record data: 'dataLen' bytes][EDC]
67
68 The LAST field of the header MUST be the Size_T record data length.
69
70 The 'expectedDataLen' is used to read the entire chunk with a SINGLE
71 IMedia::read() call. Additional reads are only issued when the stored
72 data length is larger than 'expectedDataLen', or when the single read
73 fails (e.g. the media is shorter than the expected chunk length).
74
75 Returns true and sets 'dataLen' if the chunk is valid; else false is
76 returned. NOTE: m_workBuffer is overwritten, i.e. any previous chunk
77 (or pointers into it) are invalidated.
78 */
79 bool readChunk( IMedia& media, Size_T index, Size_T headerLen, Size_T expectedDataLen, Size_T& dataLen ) noexcept;
80
81 /** Helper method that computes the EDC over the first 'len' bytes of
82 m_workBuffer (i.e. header + record data), appends the EDC to the
83 buffer, and then writes the entire chunk to 'media' at 'index' with a
84 SINGLE IMedia::write() call. When 'invalidate' is true, the EDC is
85 corrupted. The caller is required to ensure that 'len' plus the EDC
86 size does not exceed m_workBufferSize. NOTE: The EDC bytes are written
87 into m_workBuffer immediately after the first 'len' bytes.
88
89 Returns true if the write was successful, else false is returned.
90 */
91 bool writeChunk( IMedia& media, Size_T index, Size_T len, bool invalidate ) noexcept;
92
93protected:
94 /** Helper method. Encapsulates pushing 'sizeDataToPush' bytes of record
95 data, starting at 'srcData' (typically inside m_workBuffer), to the
96 record. Returns the result of IPayload::copyFrom().
97 */
98 virtual bool pushToRecord( IPayload& dstHandler, const uint8_t* srcData, Size_T sizeDataToPush );
99
100 /** Helper method. Encapsulates retrieving up to 'maxDstLen' bytes of record
101 data into 'dstBuffer' (typically inside m_workBuffer). Returns the
102 length of the data, or KIT_PERSISTENCE_SIZE_MAX on error.
103 */
104 virtual Size_T pullFromRecord( IPayload& srcHandler, uint8_t* dstBuffer, Size_T maxDstLen );
105
106 /// Helper method. Encapsulates actions that occur when there is NO VALID data
107 virtual void resetChunkOnBadData();
108
109protected:
110 /// Persistent storage media
111 IMedia& m_media;
112
113 /// CRC handler
114 Kit::Checksum::IEdc& m_crc;
115
116 /// Buffer for storing the Record's data when transferring to/from persistent storage
117 uint8_t* m_workBuffer;
118
119 /// Work buffer size (in bytes)
120 Size_T m_workBufferSize;
121};
