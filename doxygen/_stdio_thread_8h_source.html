30/** This concrete class provides the 'threading wrapper' for running a
31 TShell Command Processor. This requires that the Input/Output streams
32 be provided when the TShell is launched. The TShell runs in a dedicated
33 thread with blocking waiting semantics with respect to its input.
34
35 The TShell can be started, stopped, and then restarted multiple times.
36
37 NOTE: This class dynamically allocates memory and dynamically creates a
38 Thread. However, the memory and thread are allocated/created ONCE,
39 i.e. stopping and restarting the TShell does NOT free/re-allocate the
40 dynamic memory or destroy/re-create the thread.
41*/
42class StdioThread
43{
44public:
45 /** Constructor. The 'thisIsAStaticInstance' argument is to inform the
46 instance being create that it is being created statically (i.e. before
47 main() is entered) - which is the intended typically behavior. This
48 knowledge is used to inhibit delete/destroy-thread actions in the
49 instance's destructor. This is necessary because there is no
50 guaranteed order to when static destructor fire and as such the state
51 of the KIT Libraries static resources (e.g. mutexes) are unknown
52 which your application will crash/behavior poorly on exit.
53 */
54 StdioThread( IProcessor& shell,
55 const char* threadName = "TShell",
56 int threadPriority = KIT_SYSTEM_THREAD_PRIORITY_NORMAL + KIT_SYSTEM_THREAD_PRIORITY_LOWER,
57 bool thisIsAStaticInstance = true ) noexcept
58 : m_shell( shell )
59 , m_threadPtr( nullptr )
60 , m_name( threadName )
61 , m_runnablePtr( nullptr )
62 , m_priority( threadPriority )
63 , m_staticInstance( thisIsAStaticInstance )
64 {
65 }
66
67 /// Destructor
68 ~StdioThread();
69
70public:
71 /** This method starts the Processor. It is thread safe in that it
72 runs in the context of the calling thread - and spawns a new
73 thread for Shell/Command Processor to executing in.
74
75 When the 'enabledOnLaunch' parameter is set to true, the tshell will
76 immediately start processing input. When 'enabledOnLaunch' is set to
77 false, the thread will be created, but the tshell will not be started
78 until restartTShell() is called.
79
80 The stream parameters (infd, outfd) are required to be valid for the
81 lifetime of the StdioThread instance.
82
83 The method returns true if successful, else false is returned.
84 NOTE: If launchTShell is called more than once it will always return
85 false.
86 */
87 bool launchTShell( Kit::Io::IInput& infd,
88 Kit::Io::IOutput& outfd,
89 bool enabledOnLaunch = true ) noexcept;
90
91 /** This method is used to restart the tshell. This method can ONLY be
92 called AFTER launch has been called.
93
94 The method returns true if the tshell was successfully restarted,
95 else false is returned (e.g. launchTShell() was never called)
96 */
97 bool restartTShell() noexcept;
98
99 /** This method is used to request that the tshell be stopped AND optionally
100 that its thread be terminated.
101
102 Note: The tshell self terminates when a 'bye tshell' command was issued
103 by the user or a IO error on the tshell's IO Streams are encounter.
104 When the tshell self terminates - the thread is placed into a
105 blocking state, i.e. waiting for a call to restartTShell().
106 */
107 void pleaseStopTShell( bool stopThread = false ) noexcept;
108
109 /// This method returns true when the tshell is running
110 bool isTShellRunning() noexcept;
111
112protected:
113 /// Command Processor to run
114 IProcessor& m_shell;
115
116 /// Thread that the shell runs in
117 Kit::System::Thread* m_threadPtr;
118
119 /// Thread name for the shell
120 const char* m_name;
121
122 /// Runnable instance
123 Kit::System::IRunnable* m_runnablePtr;
124
125 /// Thread priority to run the shell
126 int m_priority;
127
128 /// How I was created
129 bool m_staticInstance;
