52/** This concrete class is a 'client' to request/trigger the 'server' to generate
53 a random number. See the README.md file for more details
54 */
55class Client : public Kit::Itc::OpenCloseSync, public Kit::System::Timer
56{
57public:
58 /// Constructor
59 Client( Kit::EventQueue::IQueue& myMbox,
60 MpFoo& mpFoo )
61 : Kit::Itc::OpenCloseSync( myMbox )
62 , Kit::System::Timer( myMbox )
63 , m_mpFoo( mpFoo )
64 , m_obFoo( myMbox )
65 , m_opened( false )
66 {
67 m_obFoo.setCallback<Client, &Client::triggerChanged>( this );
68 }
69
70public:
71 /// See Kit::Itc::OpenCloseSync
72 void request( OpenMsg& msg ) noexcept override
73 {
74 // Only 'open' when not already opened
75 if ( !m_opened )
76 {
77 // Housekeeping
78 m_opened = true;
79 m_offset = 0;
80 m_iterationCount = 0;
81
82 // Send the first server request
83 expired();
84 }
85
86 // Return the message to the sender since we are done processing the request
87 msg.returnToSender();
88 }
89
90 /// See Kit::Itc::OpenCloseSync
91 void request( CloseMsg& msg ) noexcept override
92 {
93 // Only 'close' when already opened
94 if ( m_opened )
95 {
96 // Housekeeping
97 m_opened = false;
98
99 // Make sure the all of the potential callbacks get cancelled
100 Timer::stop();
101 m_mpFoo.detach( m_obFoo );
102 }
103
104 // Return the message to the sender since we are done processing the request
105 msg.returnToSender();
106 }
107
108
109protected:
110 /// Callback for when the software timer expires
111 void expired() noexcept override
112 {
113 // Check for the Application-Exit condition
114 if ( m_iterationCount >= OPTION_EXAMPLES_DM_TRANSACTION_CLIENT_NUM_ITERATIONS )
115 {
116 Kit::System::Shutdown::success();
117 return;
118 }
119
120 // Trigger the server request and subscribe for the server's response
121 m_iterationCount++;
122 int32_t lowerLimit = m_offset;
123 int32_t upperLimit = m_offset + OPTION_EXAMPLES_DM_TRANSACTION_CLIENT_LIMIT_RANGE;
124 uint16_t seqNum = m_mpFoo.triggerRequest( upperLimit, lowerLimit );
125 m_mpFoo.attach( m_obFoo, seqNum );
126 }
127
128 /// Callback when the 'trigger' MP changes
129 void triggerChanged( MpFoo& mpThatChanged, Kit::Dm::IObserver& observer ) noexcept
130 {
131 Foo_T data;
132 if ( mpThatChanged.readAndSync( data, observer ) )
133 {
134 KIT_SYSTEM_TRACE_MSG( "main",
135 "client: RESPONSE. upper=%" PRId32 ", lower=%" PRId32 ", result=%" PRId32 ", isValid=%s",
136 data.upperLimit,
137 data.lowerLimit,
138 data.randomNumber,
139 data.isValid ? "true" : "false" );
140
141 // Cancel the callback subscription since we got the response from the server
142 mpThatChanged.detach( m_obFoo );
143
144 // "reset" the MP for next transaction
145 mpThatChanged.setInvalid();
146
147 // Start the timer for the next transaction.
148 m_offset += OPTION_EXAMPLES_DM_TRANSACTION_CLIENT_TRANSACTION_DELTA;
149 Timer::start( OPTION_EXAMPLES_DM_TRANSACTION_CLIENT_REQUEST_INTERVAL_MS );
150 }
151 }
152
153protected:
154 /// MP for the Client/Server transaction
155 MpFoo& m_mpFoo;
156
157 /// Observer callback for the MP change notification
158 Kit::Dm::ObserverCallback<MpFoo> m_obFoo;
159
160 /// Offset for upper/lower limit for the next transaction
161 int32_t m_offset;
162
163 /// Iteration count for the Client/Server transaction
164 unsigned m_iterationCount;
165
166 /// Track my open/closed
167 bool m_opened;
168};
