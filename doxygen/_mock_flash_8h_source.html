70 Result_T read( size_t srcOffset,
71 void* dstBuffer,
72 size_t numBytes ) noexcept override
73 {
74 if ( m_failNext )
75 {
76 m_failNext = false;
77 return Result_T::ERR_FAILED;
78 }
79
80 if ( dstBuffer == nullptr || numBytes == 0 )
81 {
82 return Result_T::ERR_FAILED;
83 }
84
85 if ( srcOffset + numBytes > TOTAL_SIZE )
86 {
87 return Result_T::ERR_RANGE;
88 }
89
90 memcpy( dstBuffer, m_flash + srcOffset, numBytes );
91 m_readCount++;
92 return Result_T::SUCCESS;
93 }
94
95 Result_T write( size_t dstOffset,
96 const void* srcBuffer,
97 size_t numBytes ) noexcept override
98 {
99 if ( m_failNext )
100 {
101 m_failNext = false;
102 return Result_T::ERR_FAILED;
103 }
104
105 if ( srcBuffer == nullptr || numBytes == 0 )
106 {
107 return Result_T::ERR_FAILED;
108 }
109
110 if ( dstOffset + numBytes > TOTAL_SIZE )
111 {
112 return Result_T::ERR_RANGE;
113 }
114
115 // Simulate flash write semantics: can only clear bits (AND operation)
116 const uint8_t* src = static_cast<const uint8_t*>( srcBuffer );
117 for ( size_t i = 0; i < numBytes; i++ )
118 {
119 m_flash[dstOffset + i] &= src[i];
120 }
121
122 m_writeCount++;
123 return Result_T::SUCCESS;
124 }
125
126 Result_T eraseSector( size_t sectorAddress,
127 unsigned numberOfSectors = 1 ) noexcept override
128 {
129 if ( m_failNext )
130 {
131 m_failNext = false;
132 return Result_T::ERR_FAILED;
133 }
134
135 if ( numberOfSectors == 0 )
136 {
137 return Result_T::ERR_FAILED;
138 }
139
140 if ( sectorAddress % SECTOR_SIZE != 0 )
141 {
142 return Result_T::ERR_ALIGNMENT;
143 }
144
145 // Expressed as a division so that the sector run cannot overflow size_t
146 if ( sectorAddress >= TOTAL_SIZE ||
147 ( TOTAL_SIZE - sectorAddress ) / SECTOR_SIZE < numberOfSectors )
148 {
149 return Result_T::ERR_RANGE;
150 }
151
152 memset( m_flash + sectorAddress, 0xFF, numberOfSectors * SECTOR_SIZE );
153 m_eraseCount++;
154 return Result_T::SUCCESS;
155 }
