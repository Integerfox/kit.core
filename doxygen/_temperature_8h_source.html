47/** This concrete class implements IJob interface that polls a model point for a
48 temperature value and periodically writes the value to the trace output.
49 The IJob also collects some metrics and has the option to display temperature
50 in degrees Fahrenheit or Celsius.
51
52 NOTE: Multiple of instance of the class can be created - as long as each
53 instance has a unique 'name'.
54
55 The class is intended to be example/template for an IJob - not that it does
56 anything particular noteworthy.
57 */
58class Temperature : public Kit::Job::JobBase, public Kit::System::Timer

148 /// Helper method to parse the 'command line' options
149 bool parse( char* argString ) noexcept
150 {
151 Kit::Text::Tokenizer::TextBlock args( argString );
152
153 // Default the arguments
154 m_sampleMs = OPTION_EXAMPLE_JOB_TEMPERATURE_SAMPLE_INTERVAL_MS;
155 m_displayMs = OPTION_EXAMPLE_JOB_TEMPERATURE_DISPLAY_INTERVAL_MS;
156 m_fahrenheit = OPTION_EXAMPLE_JOB_TEMPERATURE_FAHRENHEIT;
157
158 // No arguments
159 if ( args.numParameters() == 0 )
160 {
161 return true;
162 }
163
164 // Sample time
165 if ( !Kit::Text::StringTo::unsignedInt( m_sampleMs, args.getParameter( 0 ) ) )
166 {
167 return false;
168 }
169
170 // ONLY sample time is provided
171 if ( args.numParameters() == 1 )
172 {
173 return true;
174 }
175
176 // Display time
177 else
178 {
179 if ( !Kit::Text::StringTo::unsignedInt( m_displayMs, args.getParameter( 1 ) ) )
180 {
181 return false;
182 }
183 }
184
185 /// ONLY sample time & display time provided
186 if ( args.numParameters() == 2 )
187 {
188 return true;
189 }
190 else
191 {
192 // Fahrenheit/Celsius
193 if ( args.numParameters() == 3 )
194 {
195 const char* arg3 = args.getParameter( 2 );
196 if ( *arg3 == 'F' || *arg3 == 'f' )
197 {
198 m_fahrenheit = true;
199 return true;
200 }
201 else if ( *arg3 == 'C' || *arg3 == 'c' )
202 {
203 m_fahrenheit = false;
204 return true;
205 }
206 }
207 }
208
209 // If I get here the parsing failed
210 return false;
211 }
212
213 /// Timer expired callback
214 void expired( void ) noexcept override
215 {
216 // Toggle the LED to provide visual feedback on the HW
217 Bsp_toggle_debug1();
218
219 // Sample the current temperature
220 float t;
221 if ( m_mpTemperature.read( t ) )
222 {
223 m_invalidData = false;
224
225 // Metrics
226 if ( t < m_minTemp )
227 {
228 m_minTemp = t;
229 }
230 if ( t > m_maxTemp )
231 {
232 m_maxTemp = t;
233 }
234
235 // Running totals
236 m_numSamples++;
237 m_sumTemp += t;
238
239 uint32_t now = Kit::System::ElapsedTime::milliseconds();
240 if ( Kit::System::ElapsedTime::expiredMilliseconds( m_timeMarkerMs, m_displayMs, now ) || m_numSamples == 1 )
241 {
242 m_timeMarkerMs = now;
243
244 float avgTemp = m_sumTemp / m_numSamples;
245 float displayTemp = t;
246 float maxTemp = m_maxTemp;
247 float minTemp = m_minTemp;
248 char units = 'C';
249 if ( m_fahrenheit )
250 {
251 avgTemp = ( avgTemp * ( 9.0F / 5.0F ) ) + 32.0F;
252 displayTemp = ( displayTemp * ( 9.0F / 5.0F ) ) + 32.0F;
253 maxTemp = ( maxTemp * ( 9.0F / 5.0F ) ) + 32.0F;
254 minTemp = ( minTemp * ( 9.0F / 5.0F ) ) + 32.0F;
255 units = 'F';
256 }
257
258 KIT_SYSTEM_TRACE_MSG( OPTION_KIT_JOB_TRACE_SECTION,
259 "%s: %g '%c, avg: %g '%c, min: %g '%c, max: %g '%c",
260 m_name,
261 displayTemp,
262 units,
263 avgTemp,
264 units,
265 minTemp,
266 units,
267 maxTemp,
268 units );
269 }
270 }
271
272 // Only print an error message on the transition
273 else if ( !m_invalidData )
274 {
275 m_invalidData = true;
276 KIT_SYSTEM_TRACE_MSG( OPTION_KIT_JOB_TRACE_SECTION, "%s: <data invalid>", m_name );
277 }
278
279 // Restart my timer
280 Timer::start( m_sampleMs );
281 }
282
