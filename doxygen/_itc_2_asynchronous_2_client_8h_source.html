28/** This concrete class is a asynchronous ITC client that sets the server's

31class Client : public IRateResponse, public Kit::Itc::OpenCloseSync, public Kit::System::Timer

33public:
34 /// Constructor
35 Client( Kit::EventQueue::IQueue& clientsEventQueue,
36 Itc::Synchronous::IRateRequest::SAP& serverSAP,
37 uint32_t flashRatesMs[],
38 unsigned flashRatesCount,
39 Kit::System::Semaphore& shutdownSignal )
40 : Kit::Itc::OpenCloseSync( clientsEventQueue )
41 , Kit::System::Timer( clientsEventQueue )
42 , m_shutdownSignal( shutdownSignal )
43 , m_serverSAP( serverSAP )
44 , m_rateResponseMsg( *this, clientsEventQueue, serverSAP, m_ratePayload )
45 , m_flashRatesMs( flashRatesMs )
46 , m_flashRatesCount( flashRatesCount )
47 , m_maxNumberOfRequests( 0 )
48 , m_opened( false )
49 {
50 KIT_SYSTEM_ASSERT( m_flashRatesMs != nullptr );
51 }
52
53public:
54 /// See Kit::Itc::OpenCloseSync
55 void request( OpenMsg& msg ) noexcept override
56 {
57 // Only 'open' when not already opened
58 if ( !m_opened )
59 {
60 // Housekeeping
61 m_opened = true;
62 m_flashRateIndex = 0;
63
64 // Get the maximum number of requests from the message payload
65 m_maxNumberOfRequests = static_cast<int>( reinterpret_cast<size_t>( msg.getPayload().args ) );
66
67 // Send the first server request
68 sendRequest();
69 }
70
71 // Return the message to the sender since we are done processing the request
72 msg.returnToSender();
73 }
74
75 /// See Kit::Itc::OpenCloseSync
76 void request( CloseMsg& msg ) noexcept override
77 {
78 // Only 'close' when already opened
79 if ( m_opened )
80 {
81 // Housekeeping
82 m_opened = false;
83 Timer::stop();
84
85 // NOTE: The implementation is assuming that the Class is opened and
86 // closed ONLY once with respect to ITC message - but nothing
87 // prevents the application abusing these semantics. A more
88 // robust shutdown would ensure that any pending requests have
89 // been returned. See the 'Itc/AsyncCancel' example for how to
90 // properly handle this scenario.
91 m_rateResponseMsg.abandon();
92 }
93
94 // Return the message to the sender since we are done processing the request
95 msg.returnToSender();
96 }
97
98public:
99 /// Rate Response
100 void response( RateMsg& msg ) noexcept override
101 {
102 // Process the Response
103 auto& payload = msg.getPayload();
104 if ( payload.success )
105 {
106 // Run each flash rate for 10 full on/off cycles
107 Timer::start( payload.flashRateMs * 2 * 10 );
108 }
109
110 // End the application on error
111 else
112 {
113 KIT_SYSTEM_TRACE_MSG( "main", "Rate Request FAILED. (delay=%" PRIu32 " ms)", payload.flashRateMs );
114 m_shutdownSignal.signal();
115 }
116 }
117
118protected:
119 /// Timer callback
120 void expired() noexcept override
121 {
122 // Limit the number of requests
123 if ( --m_maxNumberOfRequests > 0 )
124 {
125 sendRequest();
126 }
127
128 // Shutdown the application when the maximum number of requests has been reached
129 else
130 {
131 KIT_SYSTEM_TRACE_MSG( "main", "Maximum number of requests reached." );
132 m_shutdownSignal.signal();
133 }
134 }
135
136 /// Helper method: Send the rate request to the server
137 void sendRequest() noexcept
138 {
139 // Buid the Request payload
140 auto& payload = m_rateResponseMsg.getPayload();
141 payload.flashRateMs = m_flashRatesMs[m_flashRateIndex];
142 KIT_SYSTEM_TRACE_MSG( "main", "Sending Rate Request (flashRate=%" PRIu32 " ms, idx=%" PRIu32 ")", payload.flashRateMs, m_flashRateIndex );
143
144 // Increment the flash rate index for the next request
145 incrementFlashRateIndex();
146
147 // Send the request to the server
148 m_serverSAP.post( m_rateResponseMsg.getRequestMsg() );
149 }
150
151 /// Helper
152 void incrementFlashRateIndex() noexcept
153 {
154 m_flashRateIndex = ( m_flashRateIndex + 1 ) % m_flashRatesCount;
155 }
156
157protected:
158 /// Used to signal the main thread that the client has finished its work.
159 Kit::System::Semaphore& m_shutdownSignal;
160
161 /// Server Widget's SAP (aka its fully qualified 'handle')
162 Examples::Itc::Synchronous::IRateRequest::SAP& m_serverSAP;
163
164 /// Rate Msg: Payload
165 Examples::Itc::Synchronous::IRateRequest::RatePayload_T m_ratePayload;
166
167 /// Rate Msg: Response message (which contains the Request message)
168 IRateResponse::RateMsg m_rateResponseMsg;
169
170 /// Flash rate table
171 uint32_t* m_flashRatesMs;
172
173 /// Index of the current flash rate
174 uint32_t m_flashRateIndex;
175
176 /// Flash rate table count
177 unsigned m_flashRatesCount;
178
179 /// Maximum number of requests to send
180 int m_maxNumberOfRequests;
181
182 /// Track my open/closed
183 bool m_opened;
