155 /** Atomic Read and then clear bits operation.
156
157 NOTES:
158 1. The return data value is BEFORE the clear operation.
159 2. If the MP is invalid then the model point is set to the valid
160 state AND 'dstData' is set to zero.
161 3. The returned sequence number is the AFTER the clear operation.
162 */
163 inline uint16_t readThenClearBits( ELEMTYPE& dstData,
164 ELEMTYPE maskToClear,
165 ChangeNotice_T chgNotice = ChangeNotice_T::eNORMAL,
166 LockRequest_T lockRequest = LockRequest_T::eNO_REQUEST ) noexcept
167 {
168 Kit::System::Mutex::ScopeLock criticalSection( Kit::Dm::ModelPointBase::m_modelDatabase.getMutex_() );
169 dstData = this->m_data;
170 ELEMTYPE newData = dstData & ~maskToClear;
171 return Kit::Dm::ModelPointBase::writeData( &newData, sizeof( ELEMTYPE ), chgNotice, lockRequest );
172 }
173
174 /// Atomic Read and then clear (i.e. set to 0) operation.
175 inline uint16_t readThenClear( ELEMTYPE& dstData,
176 ChangeNotice_T chgNotice = ChangeNotice_T::eNORMAL,
177 LockRequest_T lockRequest = LockRequest_T::eNO_REQUEST ) noexcept
178 {
179 return readThenClearBits( dstData, (ELEMTYPE)-1, chgNotice, lockRequest );
180 }

186 {
187 Kit::System::Mutex::ScopeLock criticalSection( Kit::Dm::ModelPointBase::m_modelDatabase.getMutex_() );
188 return this->write( this->m_data | maskToOR, chgNotice, lockRequest );
189 }

217 }

219public:
220 /// Atomic operation to set the zero indexed bit to a 1.
221 inline uint16_t setBit( uint8_t bitPosition,
222 ChangeNotice_T chgNotice = ChangeNotice_T::eNORMAL,
223 LockRequest_T lockRequest = LockRequest_T::eNO_REQUEST ) noexcept
224 {
225 Kit::System::Mutex::ScopeLock criticalSection( Kit::Dm::ModelPointBase::m_modelDatabase.getMutex_() );
226 typedef typename std::make_unsigned<ELEMTYPE>::type UnsignedElemType;
227 const uint8_t bitWidth = sizeof( UnsignedElemType ) * 8;
228 if ( bitPosition >= bitWidth )
229 {
230 return this->m_seqNum;
231 }
232
233 UnsignedElemType data = static_cast<UnsignedElemType>( this->m_data );
234 UnsignedElemType mask = static_cast<UnsignedElemType>( 1u ) << bitPosition;
235 return this->write( static_cast<ELEMTYPE>( data | mask ), chgNotice, lockRequest );
236 }

297 \endcode
298 */
299 bool setJSONVal( JsonDocument& doc ) noexcept
300 {
301 Kit::Text::FString<sizeof( OPTION_KIT_DM_MP_MAX_SIGNED_INT_TYPE ) * 2 + 2> hexString;
302 //
303 OPTION_KIT_DM_MP_MAX_SIGNED_INT_TYPE val = (typename std::make_unsigned<ELEMTYPE>::type)this->m_data;
304 JsonObject valObj = doc.createNestedObject( "val" );
305 hexString.format( "0x%" OPTION_KIT_DM_MP_MAX_INT_HEX_PRINTF_FORMAT, val );
306 valObj["dec"] = this->m_data;
307 valObj["hex"] = (char*)hexString.getString();
308 return true;
309 }

311public:
312 /** Override the Parent's implementation. See Kit::Dm::Point.
313 \code
314 Input: val: <number> OR val: "<string that can be parsed as a number>"
315 For example:
316 { "val": 123 }
317 { "val": "123" }
318 { "val": "0x7B" }
319 \endcode
320 */
321 bool fromJSON_( JsonVariant& src, LockRequest_T lockRequest, uint16_t& retSequenceNumber, Kit::Text::IString* errorMsg ) noexcept
322 {
323 // Parse a number
324 if ( src.is<ELEMTYPE>() )
325 {
326 retSequenceNumber = this->write( src.as<ELEMTYPE>(), ChangeNotice_T::eNORMAL, lockRequest );
327 return true;
328 }
329
330 // Parse as a string (e.g. "0x12" or "123")
331 if ( src.is<const char*>() )
332 {
333 // Try to parse the string as a number
334 OPTION_KIT_DM_MP_MAX_SIGNED_INT_TYPE val;
335 if ( Kit::Text::StringTo::signedInt( val, src.as<const char*>(), 0 ) )
336 {
337 retSequenceNumber = this->write( static_cast<ELEMTYPE>( val ), ChangeNotice_T::eNORMAL, lockRequest );
338 return true;
339 }
340 }
341 if ( errorMsg )
342 {
343 *errorMsg = "Invalid syntax for the 'val' key/value pair";
344 }
345 return false;
346 }
